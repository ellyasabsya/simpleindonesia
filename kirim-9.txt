<?php

namespace App\Http\Controllers;

use App\Models\Event;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;

class TicketPurchaseController extends Controller
{
    private const HOLD_MINUTES = 60;

    private function guard(Event $event): void
    {
        abort_unless(
            $event->status === Event::PUBLISHED
            && $event->registration_method === 'ticket'
            && $event->ticket_sales_open,
            404
        );
    }

    /** Metode pembayaran tiket yang benar-benar tersedia saat ini. */
    private function methods(): array
    {
        $m = [];
        if (\App\Models\Setting::manualOn() && \App\Models\Setting::bank()['bank_account_number'] !== '') {
            $m[] = 'transfer_manual';
        }

        return $m;
    }

    private function sold(int $typeId): int
    {
        return (int) DB::table('ticket_orders')->where('ticket_type_id', $typeId)
            ->where('status', '!=', 'batal')
            ->where(fn ($q) => $q->where('status', '!=', 'menunggu')->orWhere('expires_at', '>', now()))
            ->sum('quantity');
    }

    private function id($date, string $fmt = 'l, j F Y'): string
    {
        return $date->copy()->locale('id')->translatedFormat($fmt);
    }

    /** URL lama: arahkan ke halaman Semua Tiket dengan modal terbuka. */
    public function show(Event $event): RedirectResponse
    {
        $this->guard($event);

        return redirect('/tickets?beli='.$event->id);
    }

    public function data(Event $event): JsonResponse
    {
        $this->guard($event);

        $types = $event->ticketTypes()
            ->whereDate('valid_from', '<=', now()->toDateString())
            ->whereDate('valid_until', '>=', now()->toDateString())
            ->get()
            ->map(fn ($t) => [
                'id' => $t->id,
                'name' => $t->name,
                'price' => (int) $t->price,
                'remaining' => max(0, $t->quota - $this->sold($t->id)),
                'valid_from' => $this->id($event->starts_at),
                'valid_until' => $this->id($event->starts_at->copy()->addDays(max(1, (int) $t->valid_days) - 1)),
                'require_holder_details' => (bool) $t->require_holder_details,
                'support_enabled' => (bool) $t->support_enabled,
                'support_options' => json_decode((string) $t->support_options, true) ?: [],
                'max_per_order' => (int) $t->max_per_order,
            ])->values();

        return response()->json([
            'event' => [
                'id' => $event->id,
                'title' => $event->title,
                'location' => $event->location,
                'date' => $this->id($event->starts_at),
            ],
            'ticket_types' => $types,
            'fee_percent' => \App\Models\Setting::serviceFeePercent(),
            'payment' => ['methods' => $this->methods(), 'bank' => \App\Models\Setting::bank()],
        ]);
    }

    public function store(Request $request, Event $event): RedirectResponse
    {
        $this->guard($event);

        $tt = DB::table('ticket_types')->where('event_id', $event->id)->where('id', (int) $request->input('ticket_type_id'))->first(['price', 'require_holder_details', 'support_enabled', 'support_options']);
        $free = $tt !== null && (int) $tt->price === 0;
        $req = $tt !== null && (bool) $tt->require_holder_details;
        $sup = $tt !== null && (bool) $tt->support_enabled;
        $supOpts = $sup ? (json_decode((string) $tt->support_options, true) ?: []) : [];

        $data = $request->validate([
            'ticket_type_id' => ['required', 'integer'],
            'support_choice' => $sup ? ['required', 'string', \Illuminate\Validation\Rule::in($supOpts)] : ['nullable'],
            'holder_emails' => [$req ? 'required' : 'nullable', 'array', 'max:10'],
            'holder_emails.*' => [$req ? 'required' : 'nullable', 'email', 'max:120'],
            'holder_phones' => [$req ? 'required' : 'nullable', 'array', 'max:10'],
            'holder_phones.*' => [$req ? 'required' : 'nullable', 'string', 'max:30'],
            'quantity' => ['required', 'integer', 'min:1', 'max:10'],
            'buyer_name' => ['required', 'string', 'max:100'],
            'buyer_email' => ['required', 'email', 'max:120'],
            'buyer_phone' => ['required', 'string', 'max:30'],
            'holders' => ['required', 'array', 'min:1', 'max:10'],
            'holders.*' => ['required', 'string', 'max:100'],
            'payment_method' => $free ? ['nullable'] : ['required', \Illuminate\Validation\Rule::in($this->methods())],
            'payment_proof' => $free ? ['nullable'] : ['required', 'file', 'mimes:jpg,jpeg,png,webp,pdf', 'max:5120'],
        ], [
            'required' => ':attribute wajib diisi.',
            'email' => 'Format email tidak valid.',
            'payment_method.in' => 'Metode pembayaran tidak tersedia.',
            'payment_proof.mimes' => 'Bukti transfer harus JPG, PNG, WebP, atau PDF.',
            'payment_proof.max' => 'Ukuran bukti transfer maksimal 5 MB.',
        ], [
            'buyer_name' => 'Nama', 'buyer_email' => 'Email', 'buyer_phone' => 'No. telepon',
            'holders.*' => 'Nama pemegang tiket', 'ticket_type_id' => 'Jenis tiket',
            'payment_proof' => 'Bukti transfer', 'payment_method' => 'Metode pembayaran',
            'support_choice' => 'Pilihan dukungan', 'holder_emails.*' => 'Email pemegang tiket', 'holder_phones.*' => 'Telepon pemegang tiket',
        ]);

        $fail = fn (string $k, string $m) => throw ValidationException::withMessages([$k => $m]);

        [$code, $orderId] = DB::transaction(function () use ($event, $data, $fail, $request, $free, $req, $sup) {
            $type = $event->ticketTypes()->lockForUpdate()->find($data['ticket_type_id']);
            if (! $type || $type->valid_until->copy()->endOfDay()->isPast() || $type->valid_from->copy()->startOfDay()->isFuture()) {
                $fail('ticket_type_id', 'Jenis tiket tidak tersedia.');
            }
            if ($data['quantity'] > $type->max_per_order) {
                $fail('quantity', "Maksimal {$type->max_per_order} tiket per pesanan.");
            }
            if (count($data['holders']) !== (int) $data['quantity']) {
                $fail('holders', 'Jumlah pemegang tiket harus sama dengan jumlah tiket.');
            }
            if ($req && (count($data['holder_emails'] ?? []) !== (int) $data['quantity'] || count($data['holder_phones'] ?? []) !== (int) $data['quantity'])) {
                $fail('holders', 'Lengkapi email dan telepon setiap pemegang tiket.');
            }
            if ($data['quantity'] > $type->quota - $this->sold($type->id)) {
                $fail('quantity', 'Sisa tiket tidak cukup.');
            }

            $proof = $free ? null : \App\Support\ImageUploader::store($data['payment_proof'], 'payment-proofs/tiket', 'payment_proof', 1600, 'local');

            $subtotal = (int) $type->price * (int) $data['quantity'];
            $feePercent = \App\Models\Setting::serviceFeePercent();
            $fee = (int) round($subtotal * $feePercent / 100);
            $code = 'TKT-'.strtoupper(Str::random(8));

            $orderId = DB::table('ticket_orders')->insertGetId([
                'event_id' => $event->id,
                'ticket_type_id' => $type->id,
                'code' => $code,
                'user_id' => $request->user()?->id,
                'buyer_name' => trim($data['buyer_name']),
                'buyer_email' => $data['buyer_email'],
                'buyer_phone' => trim($data['buyer_phone']),
                'quantity' => (int) $data['quantity'],
                'subtotal' => $subtotal,
                'fee_percent' => $feePercent,
                'fee' => $fee,
                'total' => $subtotal + $fee,
                'holders' => json_encode(array_values($data['holders']), JSON_UNESCAPED_UNICODE),
                'status' => $free ? 'lunas' : 'menunggu_verifikasi',
                'paid_at' => $free ? now() : null,
                'payment_method' => $free ? 'gratis' : $data['payment_method'],
                'payment_proof' => $proof,
                'support_choice' => $sup ? $data['support_choice'] : null,
                'holder_details' => $req ? json_encode(array_map(fn ($e, $p) => ['email' => $e, 'phone' => $p], array_values($data['holder_emails']), array_values($data['holder_phones'])), JSON_UNESCAPED_UNICODE) : null,
                'expires_at' => null,
                'created_at' => now(), 'updated_at' => now(),
            ]);

            return [$code, $orderId];
        });

        if ($free) {
            try {
                \App\Support\TicketIssuer::issue($orderId);
            } catch (\Throwable $e) {
                report($e);
            }
        } else {
            \App\Support\AdminNotifier::send(
                'Pesanan tiket menunggu verifikasi',
                "Pesanan {$code} atas nama {$data['buyer_name']} menunggu verifikasi pembayaran.",
                '/admin/pesanan-tiket'
            );
        }

        return redirect('/tickets')->with('order_code', $code);
    }
}

<?php

namespace App\Http\Controllers\Panel;

use App\Http\Controllers\Controller;
use App\Models\Event;
use App\Models\TicketType;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class TicketSettingsController extends Controller
{
    /** Tiket yang sudah terpakai: lunas, atau menunggu pembayaran yang belum kedaluwarsa (sama dengan logika pembelian). */
    public static function sold(int $typeId): int
    {
        return (int) DB::table('ticket_orders')->where('ticket_type_id', $typeId)
            ->where('status', '!=', 'batal')
            ->where(fn ($q) => $q->where('status', '!=', 'menunggu')->orWhere('expires_at', '>', now()))
            ->sum('quantity');
    }

    private function guard(Request $request, Event $event): void
    {
        $inst = $request->attributes->get('institution');
        abort_unless($inst && $event->institution_id === $inst->id, 404);
        abort_unless($event->registration_method === 'ticket' && $event->status === Event::PUBLISHED, 404);
    }

    private function validated(Request $request, Event $event, ?TicketType $type = null): array
    {
        $end = $event->ends_at->toDateString();
        $data = $request->validate([
            'name' => ['required', 'string', 'max:60'],
            'price' => ['required', 'integer', 'min:0', 'max:100000000'],
            'quota' => ['required', 'integer', 'min:1', 'max:1000000'],
            'valid_days' => ['required', 'integer', 'min:1', 'max:365'],
            'custom_sale' => ['required', 'boolean'],
            'valid_from' => ['required_if:custom_sale,true', 'nullable', 'date'],
            'valid_until' => ['required_if:custom_sale,true', 'nullable', 'date', 'after_or_equal:valid_from', "before_or_equal:{$end}"],
            'require_holder_details' => ['required', 'boolean'],
            'max_per_order' => ['required', 'integer', 'min:1', 'max:10'],
            'support_enabled' => ['required', 'boolean'],
            'support_options' => ['nullable', 'array', 'max:50'],
            'support_options.*' => ['nullable', 'string', 'max:60'],
        ], [
            'name.required' => 'Isi nama jenis tiket.',
            'price.required' => 'Isi harga tiket (0 untuk gratis).',
            'price.integer' => 'Harga harus berupa angka.',
            'quota.required' => 'Isi kuota tiket.',
            'quota.min' => 'Kuota minimal 1.',
            'valid_days.required' => 'Isi masa berlaku tiket.',
            'valid_days.min' => 'Masa berlaku minimal 1 hari.',
            'valid_from.required_if' => 'Isi tanggal mulai jual.',
            'valid_until.required_if' => 'Isi tanggal akhir jual.',
            'valid_until.after_or_equal' => 'Tanggal akhir jual tidak boleh sebelum tanggal mulai.',
            'valid_until.before_or_equal' => 'Penjualan tidak boleh melewati akhir event ('.$end.').',
            'max_per_order.max' => 'Maksimal 10 tiket per pesanan.',
        ]);

        $data['name'] = trim($data['name']);

        $opts = collect($data['support_options'] ?? [])->map(fn ($o) => trim((string) $o))->filter()
            ->unique(fn ($o) => mb_strtolower($o))->values()->all();
        if ($data['support_enabled'] && count($opts) < 2) {
            throw \Illuminate\Validation\ValidationException::withMessages(['support_options' => 'Isi minimal 2 pilihan dukungan, satu per baris.']);
        }
        $data['support_options'] = $data['support_enabled'] ? json_encode($opts, JSON_UNESCAPED_UNICODE) : null;
        $dupe = TicketType::where('event_id', $event->id)
            ->whereRaw('lower(name) = ?', [mb_strtolower($data['name'])])
            ->when($type, fn ($q) => $q->where('id', '!=', $type->id))
            ->exists();
        if ($dupe) {
            throw \Illuminate\Validation\ValidationException::withMessages(['name' => 'Nama jenis tiket sudah dipakai di event ini.']);
        }

        if (! $data['custom_sale']) {
            $data['valid_from'] = ($type && ! $type->custom_sale) ? $type->valid_from->toDateString() : now()->toDateString();
            $data['valid_until'] = $end;
        }

        return $data;
    }

    public function toggle(Request $request, Event $event): RedirectResponse
    {
        $this->guard($request, $event);
        $data = $request->validate(['open' => ['required', 'boolean']]);

        if ($data['open'] && ! TicketType::where('event_id', $event->id)->exists()) {
            return back()->with('error', 'Tambahkan minimal satu jenis tiket dulu.');
        }

        DB::table('events')->where('id', $event->id)->update(['ticket_sales_open' => $data['open'] ? 1 : 0]);

        return back()->with('success', $data['open'] ? 'Penjualan tiket dibuka.' : 'Penjualan tiket ditutup.');
    }

    public function store(Request $request, Event $event): RedirectResponse
    {
        $this->guard($request, $event);
        $data = $this->validated($request, $event);

        TicketType::create($data + ['event_id' => $event->id]);

        return back()->with('success', 'Jenis tiket ditambahkan.');
    }

    public function update(Request $request, Event $event, TicketType $type): RedirectResponse
    {
        $this->guard($request, $event);
        abort_unless($type->event_id === $event->id, 404);
        $data = $this->validated($request, $event, $type);

        $hasOrders = DB::table('ticket_orders')->where('ticket_type_id', $type->id)->where('status', '!=', 'batal')->exists();
        if ($hasOrders && (int) $data['price'] !== (int) $type->price) {
            return back()->withErrors(['price' => 'Harga tidak bisa diubah karena sudah ada pesanan untuk jenis ini.']);
        }
        $sold = self::sold($type->id);
        if ((int) $data['quota'] < $sold) {
            return back()->withErrors(['quota' => "Kuota tidak boleh di bawah tiket yang sudah terpakai ({$sold})."]);
        }

        $type->update($data);

        return back()->with('success', 'Jenis tiket diperbarui.');
    }

    public function destroy(Request $request, Event $event, TicketType $type): RedirectResponse
    {
        $this->guard($request, $event);
        abort_unless($type->event_id === $event->id, 404);

        if (DB::table('ticket_orders')->where('ticket_type_id', $type->id)->exists()) {
            return back()->with('error', 'Jenis tiket yang sudah punya pesanan tidak bisa dihapus.');
        }
        if (TicketType::where('event_id', $event->id)->count() <= 1) {
            return back()->with('error', 'Event harus punya minimal satu jenis tiket.');
        }

        $type->delete();

        return back()->with('success', 'Jenis tiket dihapus.');
    }
}

<?php

namespace App\Http\Controllers;

use Carbon\Carbon;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Inertia\Inertia;
use Inertia\Response;

class TicketViewController extends Controller
{
    private function date(?string $d): string
    {
        return $d ? Carbon::parse($d)->locale('id')->translatedFormat('l, j F Y') : '-';
    }

    /** Halaman tiket per kode. Kode acak 24 karakter, tanpa data pribadi selain nama pemegang. */
    public function show(string $code): Response
    {
        $t = DB::table('ticket_holders as h')
            ->join('ticket_orders as o', 'o.id', '=', 'h.ticket_order_id')
            ->join('events as e', 'e.id', '=', 'h.event_id')
            ->join('ticket_types as tt', 'tt.id', '=', 'h.ticket_type_id')
            ->where('h.code', $code)
            ->where('o.status', 'lunas')
            ->first(['h.code', 'h.holder_name', 'h.checked_in_at', 'e.title', 'e.location', 'e.starts_at', 'tt.name as type_name']);
        abort_unless($t, 404);

        return Inertia::render('tiket-detail', ['ticket' => [
            'code' => $t->code,
            'holder_name' => $t->holder_name,
            'checked_in' => (bool) $t->checked_in_at,
            'event_title' => $t->title,
            'location' => $t->location,
            'date' => $this->date($t->starts_at),
            'type_name' => $t->type_name,
        ]]);
    }

    public function mine(Request $request): Response
    {
        $user = $request->user();

        $orders = DB::table('ticket_orders as o')
            ->join('events as e', 'e.id', '=', 'o.event_id')
            ->where('o.status', 'lunas')
            ->where(function ($q) use ($user) {
                $q->where('o.user_id', $user->id);
                if ($user->email_verified_at) {
                    $q->orWhere('o.buyer_email', $user->email);
                }
            })
            ->orderByDesc('o.id')
            ->get(['o.id', 'o.code', 'e.title', 'e.location', 'e.starts_at']);

        $holders = DB::table('ticket_holders')->whereIn('ticket_order_id', $orders->pluck('id'))
            ->get(['ticket_order_id', 'code', 'holder_name', 'checked_in_at'])->groupBy('ticket_order_id');

        return Inertia::render('tiket-saya', ['orders' => $orders->map(fn ($o) => [
            'code' => $o->code,
            'event_title' => $o->title,
            'location' => $o->location,
            'date' => $this->date($o->starts_at),
            'tickets' => ($holders[$o->id] ?? collect())->map(fn ($h) => [
                'code' => $h->code, 'holder_name' => $h->holder_name, 'checked_in' => (bool) $h->checked_in_at,
            ])->values(),
        ])->values()]);
    }
}

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use Carbon\Carbon;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use Inertia\Inertia;
use Inertia\Response;

class TicketOrderController extends Controller
{
    public function index(Request $request): Response
    {
        $status = $request->query('status', 'menunggu_verifikasi');
        $status = in_array($status, ['all', 'menunggu_verifikasi', 'lunas', 'batal', 'menunggu'], true) ? $status : 'menunggu_verifikasi';
        $q = trim((string) $request->query('q', ''));

        $counts = DB::table('ticket_orders')->selectRaw('status, count(*) as c')->groupBy('status')->pluck('c', 'status');

        $orders = DB::table('ticket_orders')
            ->join('events', 'events.id', '=', 'ticket_orders.event_id')
            ->join('ticket_types', 'ticket_types.id', '=', 'ticket_orders.ticket_type_id')
            ->when($status !== 'all', fn ($w) => $w->where('ticket_orders.status', $status))
            ->when($q !== '', fn ($w) => $w->where(function ($s) use ($q) {
                $s->where('ticket_orders.code', 'like', "%{$q}%")
                    ->orWhere('ticket_orders.buyer_name', 'like', "%{$q}%")
                    ->orWhere('ticket_orders.buyer_email', 'like', "%{$q}%")
                    ->orWhere('events.title', 'like', "%{$q}%");
            }))
            ->orderByDesc('ticket_orders.id')
            ->paginate(10, [
                'ticket_orders.id', 'ticket_orders.code', 'ticket_orders.buyer_name', 'ticket_orders.buyer_email',
                'ticket_orders.buyer_phone', 'ticket_orders.quantity', 'ticket_orders.subtotal', 'ticket_orders.fee',
                'ticket_orders.total', 'ticket_orders.status', 'ticket_orders.payment_proof', 'ticket_orders.created_at',
                'events.title as event_title', 'ticket_types.name as type_name',
            ])
            ->withQueryString()
            ->through(fn ($o) => [
                'id' => $o->id,
                'code' => $o->code,
                'buyer_name' => $o->buyer_name,
                'buyer_email' => $o->buyer_email,
                'buyer_phone' => $o->buyer_phone,
                'quantity' => (int) $o->quantity,
                'subtotal' => (int) $o->subtotal,
                'fee' => (int) $o->fee,
                'total' => (int) $o->total,
                'status' => $o->status,
                'event_title' => $o->event_title,
                'type_name' => $o->type_name,
                'proof' => $o->payment_proof ? [
                    'url' => "/admin/pesanan-tiket/{$o->id}/bukti",
                    'is_pdf' => strtolower(pathinfo($o->payment_proof, PATHINFO_EXTENSION)) === 'pdf',
                ] : null,
                'created_at' => Carbon::parse($o->created_at)->locale('id')->translatedFormat('j M Y, H:i'),
            ]);

        return Inertia::render('admin/pesanan-tiket', [
            'orders' => $orders->items(),
            'pg' => [
                'current' => $orders->currentPage(),
                'last' => $orders->lastPage(),
                'total' => $orders->total(),
                'from' => $orders->firstItem(),
                'to' => $orders->lastItem(),
            ],
            'status' => $status,
            'q' => $q,
            'counts' => [
                'menunggu_verifikasi' => (int) ($counts['menunggu_verifikasi'] ?? 0),
                'lunas' => (int) ($counts['lunas'] ?? 0),
                'batal' => (int) ($counts['batal'] ?? 0),
                'menunggu' => (int) ($counts['menunggu'] ?? 0),
            ],
        ]);
    }

    public function approve(int $order): RedirectResponse
    {
        $n = DB::table('ticket_orders')->where('id', $order)->where('status', 'menunggu_verifikasi')
            ->update(['status' => 'lunas', 'paid_at' => now(), 'updated_at' => now()]);
        if ($n === 1) {
            \App\Support\TicketIssuer::issue($order);
        }

        return $n === 1
            ? back()->with('success', 'Pesanan disetujui dan tiket sudah diterbitkan.')
            : back()->with('error', 'Pesanan sudah diproses sebelumnya. Muat ulang halaman.');
    }

    public function reject(int $order): RedirectResponse
    {
        DB::table('ticket_orders')->where('id', $order)->where('status', 'menunggu_verifikasi')
            ->update(['status' => 'batal', 'updated_at' => now()]);

        return back()->with('success', 'Pesanan ditolak.');
    }

    public function proof(int $order)
    {
        $path = DB::table('ticket_orders')->where('id', $order)->value('payment_proof');
        abort_unless($path && Storage::disk('local')->exists($path), 404);

        return Storage::disk('local')->response($path);
    }
}

    const [q, setQ] = useState('');

    });

import { Check, Loader2, X } from 'lucide-react';
import { useEffect, useState } from 'react';
import { createPortal } from 'react-dom';

type TType = { id: number; name: string; price: number; remaining: number; valid_from: string; valid_until: string; max_per_order: number; require_holder_details: boolean; support_enabled: boolean; support_options: string[] };
type Data = {
    event: { id: number; title: string; location: string | null; date: string };
    ticket_types: TType[];
    fee_percent: number;
    payment: { methods: string[]; bank: { bank_name: string; bank_account_number: string; bank_account_holder: string } };
};

const rp = (n: number) => `Rp${new Intl.NumberFormat('id-ID').format(n)}`;
const inp = 'w-full rounded-xl border border-slate-200 bg-white px-4 py-3 text-sm outline-none focus:border-indigo-500 focus:ring-4 focus:ring-indigo-500/15 dark:border-white/10 dark:bg-white/5 dark:text-white';

export default function TicketBuyModal({ eventId, onClose }: { eventId: number; onClose: () => void }) {
    const [info, setInfo] = useState<Data | null>(null);
    const [failed, setFailed] = useState(false);
    const [done, setDone] = useState(false);
    const [same, setSame] = useState(true);

    const { data, setData, post, processing, errors, transform } = useForm({
        ticket_type_id: 0,
        quantity: 1,
        buyer_name: '',
        buyer_email: '',
        buyer_phone: '',
        holders: [''],
        holder_emails: [''] as string[],
        support_choice: '',
        holder_phones: [''] as string[],
        payment_method: 'transfer_manual',
        payment_proof: null as File | null,
    });

    useEffect(() => {
        fetch(`/events/${eventId}/beli-data`, { headers: { Accept: 'application/json' } })
            .then((r) => {
                if (r.status === 401 || r.status === 419) {
                    window.location.assign(`/tickets?beli=${eventId}`);
                    return new Promise<never>(() => {});
                }
                return r.ok ? r.json() : Promise.reject();
            })
            .then((d: Data) => {
                setInfo(d);
                if (d.ticket_types[0]) setData('ticket_type_id', d.ticket_types[0].id);
            })
            .catch(() => setFailed(true));
        // eslint-disable-next-line react-hooks/exhaustive-deps
    }, [eventId]);

    useEffect(() => {
        const k = (e: KeyboardEvent) => e.key === 'Escape' && onClose();
        window.addEventListener('keydown', k);
        document.body.style.overflow = 'hidden';
        return () => {
            window.removeEventListener('keydown', k);
            document.body.style.overflow = '';
        };
    }, [onClose]);

    const type = info?.ticket_types.find((t) => t.id === data.ticket_type_id);
    const max = type ? Math.max(1, Math.min(type.max_per_order, type.remaining)) : 1;
    const feePct = info?.fee_percent ?? 10;
    const subtotal = (type?.price ?? 0) * data.quantity;
    const fee = Math.round((subtotal * feePct) / 100);
    const holders = Array.from({ length: data.quantity }, (_, i) => (i === 0 && same ? data.buyer_name : data.holders[i] ?? ''));
    const needDetail = !!type?.require_holder_details;
    const hEmails = Array.from({ length: data.quantity }, (_, i) => (i === 0 && same ? data.buyer_email : data.holder_emails[i] ?? ''));
    const hPhones = Array.from({ length: data.quantity }, (_, i) => (i === 0 && same ? data.buyer_phone : data.holder_phones[i] ?? ''));
    const setHolderField = (k: 'holder_emails' | 'holder_phones', i: number, v: string) => setData(k, Array.from({ length: data.quantity }, (_, j) => (j === i ? v : data[k][j] ?? '')));
    const err = (k: string) => (errors as Record<string, string>)[k];

    const setQty = (q: number) => {
        const n = Math.max(1, Math.min(max, q));
        setData((d) => ({ ...d, quantity: n, holders: Array.from({ length: n }, (_, i) => d.holders[i] ?? ''), holder_emails: Array.from({ length: n }, (_, i) => d.holder_emails[i] ?? ''), holder_phones: Array.from({ length: n }, (_, i) => d.holder_phones[i] ?? '') }));
    };
    const setHolder = (i: number, v: string) => setData('holders', Array.from({ length: data.quantity }, (_, j) => (j === i ? v : data.holders[j] ?? '')));

    const submit = () => {
        transform((d) => ({ ...d, holders, holder_emails: needDetail ? hEmails : [], holder_phones: needDetail ? hPhones : [], support_choice: type?.support_enabled ? d.support_choice : '' }));
        post(`/events/${eventId}/beli`, { forceFormData: true, preserveScroll: true, preserveState: true, onSuccess: () => setDone(true) });
    };

    return createPortal(
        <div className="fixed inset-0 z-[9999] flex items-center justify-center bg-slate-900/60 p-4 backdrop-blur-sm" onMouseDown={(e) => e.target === e.currentTarget && onClose()}>
            <div role="dialog" aria-modal="true" className="flex max-h-[92vh] w-full max-w-2xl flex-col overflow-hidden rounded-3xl bg-white text-slate-900 shadow-2xl dark:bg-neutral-900 dark:text-white">
                <div className="flex items-start justify-between gap-4 border-b border-slate-100 px-6 py-5 dark:border-white/10">
                    <div>
                        <h2 className="text-xl font-extrabold">Beli Tiket</h2>
                        <p className="text-sm text-slate-500">{info?.event.title ?? 'Memuat...'}</p>
                    </div>
                    <button onClick={onClose} aria-label="Tutup" className="rounded-full p-1.5 text-slate-400 hover:bg-slate-100 dark:hover:bg-white/10"><X className="h-5 w-5" /></button>
                </div>

                {failed || (info && !type && !done) ? (
                    <div className="p-10 text-center text-sm text-slate-600">
                        Tiket untuk event ini tidak tersedia.
                        <div><button onClick={onClose} className="mt-4 font-bold text-indigo-700">Kembali</button></div>
                    </div>
                ) : !info ? (
                    <p className="p-10 text-center text-sm text-slate-400">Memuat...</p>
                ) : done ? (
                    <div className="p-10 text-center">
                        <p className="text-lg font-extrabold">Pesanan berhasil dibuat</p>
                        {subtotal === 0 && <p className="mt-2 text-sm font-bold text-emerald-700">Tiket gratis sudah diterbitkan dan dikirim ke email Anda.</p>}
                        <p className={`mt-2 text-sm text-slate-500 ${subtotal === 0 ? 'hidden' : ''}`}>Bukti transfer diterima. Setelah admin menyetujui, tiket (QR code) otomatis dikirim ke email Anda dan muncul di menu Tiket Saya di dashboard jika Anda login.</p>
                        <button onClick={onClose} className="mt-6 rounded-xl bg-gradient-to-r from-pink-600 to-orange-500 px-6 py-3 font-bold text-white">Tutup</button>
                    </div>
                ) : (
                    <>
                        <div className="flex-1 overflow-y-auto px-6 py-5">
                            <div className="space-y-2 rounded-2xl border border-sky-100 bg-sky-50/60 p-4 text-sm dark:border-white/10 dark:bg-white/5">
                                {([
                                    ['Tanggal', info.event.date],
                                    ['Lokasi', info.event.location ?? '-'],
                                ] as const).map(([k, v]) => (
                                    <div key={k} className="flex justify-between gap-4"><span className="text-slate-500">{k}</span><span className="text-right font-bold">{v}</span></div>
                                ))}
                            </div>

                            <h3 className="mt-6 text-base font-bold">{info.ticket_types.length > 1 ? 'Pilih Jenis Tiket' : 'Jenis Tiket'}</h3>
                            {info.ticket_types.length > 1 && <p className="mb-3 text-xs text-slate-500">Event ini punya {info.ticket_types.length} jenis tiket. Pilih salah satu.</p>}
                            <div className={`mt-3 grid gap-3 ${info.ticket_types.length > 1 ? 'sm:grid-cols-2' : ''}`}>
                                {info.ticket_types.map((t) => {
                                    const sel = t.id === data.ticket_type_id;
                                    const out = t.remaining <= 0;
                                    return (
                                        <button

                                            className={`relative rounded-2xl border-2 p-4 text-left transition disabled:cursor-not-allowed disabled:opacity-45 ${sel ? 'border-indigo-500 bg-indigo-50 ring-4 ring-indigo-500/15 dark:bg-indigo-500/10' : 'border-slate-200 hover:border-indigo-300 dark:border-white/10'}`}
                                        >
                                            <span className={`absolute right-3 top-3 flex h-6 w-6 items-center justify-center rounded-full border-2 ${sel ? 'border-indigo-500 bg-indigo-500 text-white' : 'border-slate-300 dark:border-white/20'}`}>{sel && <Check className="h-3.5 w-3.5" />}</span>

Route::get('/tiket-data', function () {
    $events = \App\Models\Event::where('status', \App\Models\Event::PUBLISHED)
        ->where('registration_method', 'ticket')
        ->where('ticket_sales_open', true)
        ->whereHas('ticketTypes')
        ->with('ticketTypes')
        ->orderBy('starts_at')
        ->get()
        ->map(function ($e) {
            $cheapest = $e->ticketTypes->sortBy('price')->first();
            return [
                'id' => $e->id,
                'href' => '/events/'.$e->id.'/beli',
                'title' => $e->title,
                'poster' => $e->posterUrl(),
                'starts_at' => $e->starts_at->translatedFormat('j M Y'),
                'ends_at' => $e->ends_at->translatedFormat('j M Y'),
                'location' => $e->location ?? '-',
                'category' => \App\Http\Controllers\Panel\EventController::CATEGORIES[$e->category] ?? null,
                'ticket_name' => $cheapest->name,
                'price' => (int) $cheapest->price,
                'valid_until' => $cheapest->valid_until->translatedFormat('j M Y'),
                'available' => $e->ticketTypes->sum('quota') > 0
                    && $e->ticketTypes->max('valid_until')->endOfDay()->isFuture(),
            ];
        })->values();

    return response()->json($events);
});
