<?php

namespace App\Http\Controllers;

use App\Models\Event;
use Illuminate\Http\Request;
use Inertia\Inertia;
use Inertia\Response;

class EventDetailController extends Controller
{
    public function __invoke(Request $request, string $key, HomeController $home): Response
    {
        $event = Event::query()
            ->where(fn ($q) => $q->where('status', Event::PUBLISHED)
                ->orWhereIn('parent_id', Event::where('status', Event::PUBLISHED)->select('id')))
            ->where(fn ($q) => $q->where('slug', $key)
                ->when(ctype_digit($key), fn ($w) => $w->orWhere('id', (int) $key)))
            ->with(['institution', 'children' => fn ($q) => $q->orderBy('sub_order')->orderBy('id')])
            ->firstOrFail();

        $parent = $event->parent_id ? Event::find($event->parent_id) : null;

        $labelsConst = '\App\Http\Controllers\Panel\EventController::CATEGORIES';
        $labels = defined($labelsConst) ? constant($labelsConst) : [];

        $contact = array_filter([
            'name' => $event->contact_name,
            'position' => $event->contact_position,
            'phone' => $event->contact_phone,
            'email' => $event->contact_email,
        ]);

        $userId = $request->user()?->id;

        $all = collect([$event])->concat($event->children);
        $eventIds = $all->pluck('id')->all();
        $cats = \Illuminate\Support\Facades\DB::table('event_participant_categories')->whereIn('event_id', $eventIds)->get()->groupBy('event_id');
        $regs = \App\Models\EventRegistration::query()->with('user:id,name')->whereIn('event_id', $eventIds)->get();

        $branches = collect();
        foreach ($all as $ev) {
            $isChild = $ev->id !== $event->id;
            $evCats = $cats->get($ev->id, collect());
            $evRegs = $regs->where('event_id', $ev->id);

            $push = function ($key, $title, $subtitle, $quota, $mine, $extra = []) use (&$branches) {
                $ok = $mine->where('status', 'disetujui');
                $branches->push([
                    'id' => $key,
                    'title' => $title,
                    'subtitle' => $subtitle,
                    'quota' => $quota,
                    'registered' => $mine->count(),
                    'approved' => $ok->count(),
                    'info' => $extra,
                    'names' => $ok->map(fn ($r) => ['name' => $r->user?->name ?? 'Peserta', 'school' => $r->school])->values(),
                ]);
            };

            if ($evCats->isNotEmpty()) {
                foreach ($evCats as $cat) {
                    $prizes = json_decode($cat->prizes ?? '[]', true) ?: [];
                    $push('c'.$cat->id,
                        $cat->competition_name ?: ($isChild ? $ev->title : $cat->name),
                        $cat->competition_name ? $cat->name : ($isChild ? $cat->name : null),
                        (int) ($cat->quota ?? 0),
                        $evRegs->where('participant_category_id', $cat->id),
                        ['min_age' => $cat->min_age, 'max_age' => $cat->max_age, 'requirements' => $cat->requirements, 'prizes' => $prizes]);
                }
            } elseif ((int) $ev->participant_quota > 0) {
                $push('e'.$ev->id, $isChild ? $ev->title : 'Kuota peserta', null, (int) $ev->participant_quota, $evRegs);
            }
        }

        $participants = [
            'registered' => $regs->count(),
            'approved' => $regs->where('status', 'disetujui')->count(),
            'branches' => $branches->values(),
        ];

        return Inertia::render('event-detail', [
            'event' => array_merge($home->card($event), [
                'description' => $event->description,
                'contact' => $contact ?: null,
                'juknis' => $event->juknisUrl(),
                'participants' => $participants,
                'has_competition' => $all->contains(fn (Event $ev) => in_array($ev->category, ['perlombaan', 'olahraga'], true)),
                'parent' => $parent ? ['title' => $parent->title, 'slug' => $parent->slug] : null,
                'siblings' => $parent ? $parent->children()->orderBy('sub_order')->orderBy('id')->get()
                    ->map(fn (Event $x) => ['id' => $x->id, 'title' => $x->title, 'slug' => $x->slug])->values() : [],
                'registration_fee' => (int) ($event->registration_fee ?: ($event->children->pluck('registration_fee')->filter()->min() ?? 0)),
                'registration_method' => $event->registration_method
                    ?: (($m = $event->children->pluck('registration_method')->filter()->unique())->count() === 1 ? $m->first() : null),
                'children' => $event->children->map(fn (Event $c) => [
                    'id' => $c->id,
                    'title' => $c->title,
                    'category' => $c->category ? ($labels[$c->category] ?? ucfirst($c->category)) : null,
                    'location' => $c->location,
                    'starts_at' => $c->starts_at?->toDateString(),
                    'ends_at' => $c->ends_at?->toDateString(),
                    'juknis' => $c->juknisUrl(),
                    'registration_fee' => (int) $c->registration_fee,
                    'slug' => $c->slug,
                    'poster' => $c->posterUrl() ?: $event->posterUrl(),
                    'quota' => (int) \Illuminate\Support\Facades\DB::table('event_participant_categories')->where('event_id', $c->id)->sum('quota'),
                    'registered' => \App\Models\EventRegistration::where('event_id', $c->id)->count(),
                ])->values(),
                'comments' => $event->comments()
                    ->with('user:id,name')
                    ->latest()
                    ->limit(50)
                    ->get()
                    ->map(fn ($c) => [
                        'id' => $c->id,
                        'body' => $c->body,
                        'author' => $c->user?->name ?? 'Pengguna',
                        'created_at' => $c->created_at->toIso8601String(),
                        'mine' => $userId !== null && $userId === $c->user_id,
                        'reported' => $c->reported_at !== null,
                    ])->values(),
            ]),
        ]);
    }
}

function waLink(phone: string) {
    const d = phone.replace(/\D/g, '');
    return `https://wa.me/${d.startsWith('0') ? `62${d.slice(1)}` : d}`;
}

function initials(name: string) {
    return name.split(/\s+/).filter(Boolean).slice(0, 2).map((w) => w[0]?.toUpperCase()).join('');
}

function dayCount(start: string | null, end: string | null) {
    if (!start) return null;
    if (!end) return 1;
    const diff = Math.round((new Date(`${end}T00:00:00`).getTime() - new Date(`${start}T00:00:00`).getTime()) / 86400000);
    return Math.max(diff, 0) + 1;
}

function Comments({ event }: { event: Detail }) {
    const { user } = useAuth();
    const { data, setData, post, processing, errors, reset } = useForm({ body: '' });

    const submit = (e: React.FormEvent) => {
        e.preventDefault();
        post(`/events/${event.id}/komentar`, { preserveScroll: true, onSuccess: () => reset() });
    };

    const remove = (id: number) => {
        if (confirm('Hapus komentar ini?')) router.delete(`/komentar/${id}`, { preserveScroll: true });
    };

    const report = (id: number) => {
        if (confirm('Laporkan komentar ini ke admin?')) router.post(`/komentar/${id}/lapor`, {}, { preserveScroll: true });
    };

    return (
        <section className="rounded-3xl border border-slate-200/70 bg-white p-6 shadow-sm md:p-8 dark:border-white/10 dark:bg-slate-900">
            <h2 className="text-xl font-bold">Komentar</h2>
            <p className="mt-1 text-sm text-slate-600 dark:text-slate-300">Ada pertanyaan soal event ini? Tanyakan di sini.</p>

            {user ? (
                <form onSubmit={submit} className="mt-5">
                    <textarea
                        value={data.body}
                        onChange={(e) => setData('body', e.target.value)}
                        maxLength={1000}
                        rows={4}
                        placeholder="Tulis komentar..."
                        className="w-full resize-y rounded-2xl border border-slate-200 bg-white px-5 py-4 text-sm text-slate-900 outline-none transition focus:border-indigo-500 focus:ring-4 focus:ring-indigo-500/15 dark:border-white/10 dark:bg-slate-950/60 dark:text-white"
                    />
                    {errors.body && <p className="mt-2 text-sm font-medium text-red-600">{errors.body}</p>}
                    <div className="mt-3 flex items-center justify-between">
                        <span className="text-xs text-slate-500 dark:text-slate-400">{data.body.length}/1000</span>
                        <button
                            type="submit"
                            disabled={processing || data.body.trim() === ''}
                            className="rounded-xl bg-indigo-900 px-6 py-2.5 text-sm font-bold text-white shadow transition hover:bg-indigo-800 disabled:cursor-not-allowed disabled:opacity-50 dark:bg-indigo-500 dark:hover:bg-indigo-400"
                        >
                            Kirim
                        </button>
                    </div>
                </form>
            ) : (
                <div className="mt-5 flex flex-col items-start gap-3 rounded-2xl bg-slate-50 p-5 sm:flex-row sm:items-center sm:justify-between dark:bg-white/5">
                    <p className="text-sm text-slate-600 dark:text-slate-300">Masuk terlebih dahulu untuk menulis komentar.</p>
                    <Link href="/login" className="rounded-xl bg-indigo-900 px-5 py-2.5 text-sm font-bold text-white shadow transition hover:bg-indigo-800">
                        Masuk
                    </Link>
                </div>
            )}

            <div className="mt-8">
                {event.comments.length === 0 ? (
                    <p className="text-sm text-slate-600 dark:text-slate-300">Belum ada komentar. Jadilah yang pertama.</p>
                ) : (
                    <ul className="space-y-5">
                        {event.comments.map((c) => (
                            <li key={c.id} className="flex gap-3">
                                <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-gradient-to-br from-blue-600 to-indigo-900 text-xs font-bold text-white">

    const hasHadiah = !!(event as unknown as { has_competition?: boolean }).has_competition && (parts?.branches ?? []).some((b) => b.info);

            /* dibatalkan pengguna */
        }

                : 'text-slate-600 hover:text-indigo-900 dark:text-slate-300 dark:hover:text-white'
        }`;

        <>
            <Head title={`${event.title} - SIMPLE`} />

function xsrf() {
    const m = document.cookie.match(/(?:^|; )XSRF-TOKEN=([^;]*)/);
    return m ? decodeURIComponent(m[1]) : '';
}

function SupportEditor({ value, onChange }: { value: string; onChange: (v: string) => void }) {
    const items = value.split('\n').map((o) => o.trim()).filter(Boolean);
    const [draft, setDraft] = useState('');
    const push = (raw: string[]) => {
        const next = [...items];
        for (const r of raw.map((x) => x.trim()).filter(Boolean)) {
            if (next.length >= 50) break;
            if (!next.some((i) => i.toLowerCase() === r.toLowerCase())) next.push(r.slice(0, 60));
        }
        onChange(next.join('\n'));
    };
    const add = () => { push([draft]); setDraft(''); };
    const remove = (i: number) => onChange(items.filter((_, j) => j !== i).join('\n'));
    const ok = items.length >= 2;

    return (
        <div className="mt-4">
            <div className="flex items-center justify-between">
                <span className={lbl}>Daftar pilihan</span>
                <span className={`rounded-full px-2.5 py-0.5 text-xs font-bold ${ok ? 'bg-emerald-100 text-emerald-800' : 'bg-amber-100 text-amber-800'}`}>{items.length} pilihan</span>
            </div>

            <div className="mt-2 flex gap-2">
                <input
                    value={draft}
                    maxLength={60}
                    onChange={(e) => setDraft(e.target.value)}
                    onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); add(); } }}
                    onPaste={(e) => {
                        const t = e.clipboardData.getData('text');
                        if (t.includes('\n')) { e.preventDefault(); push(t.split('\n')); setDraft(''); }
                    }}
                    placeholder="Ketik nama pilihan, lalu tekan Enter"
                    className={`${inp} !mt-0 flex-1`}
                />
                <button type="button" onClick={add} disabled={draft.trim() === ''} className="inline-flex items-center gap-1.5 rounded-2xl bg-indigo-700 px-4 text-sm font-extrabold text-white transition hover:bg-indigo-600 disabled:cursor-not-allowed disabled:opacity-40"><Plus className="h-4 w-4" /> Tambah</button>
            </div>

            {items.length === 0 ? (
                <div className="mt-3 rounded-2xl border-2 border-dashed border-slate-200 px-4 py-6 text-center text-sm text-slate-500 dark:border-white/10">Belum ada pilihan. Contoh: nama peserta, tim, atau kategori.</div>
            ) : (
                <ol className="mt-3 space-y-2">
                    {items.map((o, i) => (
                        <li key={o} className="flex items-center gap-3 rounded-2xl border border-slate-200 bg-white px-3 py-2 dark:border-white/10 dark:bg-white/5">
                            <span className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-indigo-100 text-xs font-extrabold text-indigo-800">{i + 1}</span>
                            <span className="min-w-0 flex-1 truncate text-sm font-bold">{o}</span>
                            <button type="button" aria-label={`Hapus ${o}`} onClick={() => remove(i)} className="flex h-8 w-8 items-center justify-center rounded-xl text-rose-500 transition hover:bg-rose-50 dark:hover:bg-rose-500/10"><Trash2 className="h-4 w-4" /></button>
                        </li>
                    ))}
                </ol>
            )}

            <p className={`mt-2 text-xs ${ok ? 'text-emerald-700' : 'text-amber-700'}`}>{ok ? 'Satu tiket = satu dukungan untuk pilihan yang dipilih pembeli.' : 'Tambahkan minimal 2 pilihan agar tiket bisa disimpan.'}</p>
        </div>
    );
}

type Row = { order: string; type: string; buyer: string; email: string; status: string; name: string; code: string | null; reg_id: string | null; contact?: string | null; support?: string | null };
type Props = {
    eventId: number;
    summary: { orders: number; sold: number; pending: number };
    types: string[];
    rows: Row[];
};

const STATUS: Record<string, { label: string; style: string }> = {
    lunas: { label: 'Terdaftar', style: 'bg-emerald-100 text-emerald-800' },
    menunggu: { label: 'Menunggu verifikasi', style: 'bg-amber-100 text-amber-800' },
    menunggu_verifikasi: { label: 'Menunggu verifikasi', style: 'bg-amber-100 text-amber-800' },
    batal: { label: 'Dibatalkan', style: 'bg-slate-200 text-slate-700' },
    dibatalkan: { label: 'Dibatalkan', style: 'bg-slate-200 text-slate-700' },
    kedaluwarsa: { label: 'Kedaluwarsa', style: 'bg-slate-200 text-slate-700' },
    ditolak: { label: 'Ditolak', style: 'bg-rose-100 text-rose-800' },
};
const card = 'rounded-3xl border border-white/70 bg-white/80 shadow-sm backdrop-blur dark:border-white/10 dark:bg-white/5';
const field = 'rounded-xl border border-slate-200 bg-white px-3 py-2.5 text-sm outline-none focus:border-indigo-400 focus:ring-2 focus:ring-indigo-400/30 dark:border-white/10 dark:bg-white/5';

export default function Registrasi({ eventId, summary, types, rows }: Props) {
    const [q, setQ] = useState('');
    const [type, setType] = useState('');
    const [status, setStatus] = useState('');

    const list = useMemo(() => {
        const k = q.trim().toLowerCase();
        return rows.filter((r) =>
            (!type || r.type === type) &&
            (!status || r.status === status) &&
            (!k || [r.reg_id ?? '', r.name, r.buyer, r.email, r.order, r.code ?? ''].some((v) => v.toLowerCase().includes(k))),
        );
    }, [rows, q, type, status]);

    const statuses = Array.from(new Set(rows.map((r) => r.status)));
    const stat = (t: string, v: number, s: string) => (
        <div className={`${card} p-5`}>
            <div className="text-xs font-extrabold uppercase tracking-wider text-slate-500">{t}</div>
            <div className="mt-2 text-3xl font-extrabold">{v}</div>
            <div className="mt-1 text-xs text-slate-500">{s}</div>
        </div>
    );

    return (
        <PanelLayout title="Registrasi">
            <div className="space-y-5">
                <div className={`${card} flex flex-wrap items-center justify-between gap-4 p-6`}>
                    <div className="flex items-start gap-4">
                        <span className="flex h-12 w-12 shrink-0 items-center justify-center rounded-2xl bg-gradient-to-br from-indigo-600 to-violet-600 text-white"><Ticket className="h-6 w-6" /></span>

                            <p className="mt-1 max-w-2xl text-sm text-slate-600 dark:text-slate-300">Pada paket ini peserta tidak mengisi formulir. Mereka terdaftar saat membeli tiket. Setiap pemegang tiket di bawah adalah peserta event.</p>
                        </div>
                    </div>
                    <div className="flex flex-wrap gap-2">
                        <Link href={`/panel/event/${eventId}/tiket`} className="rounded-xl bg-indigo-900 px-4 py-2.5 text-sm font-bold text-white hover:bg-indigo-800">Atur jenis & harga tiket</Link>
                    </div>
                </div>

                <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
                    {stat('Pesanan', summary.orders, 'Total pesanan tiket')}
                    {stat('Tiket terjual', summary.sold, 'Pembayaran lunas')}
                    {stat('Menunggu verifikasi', summary.pending, 'Belum disetujui admin')}
                </div>

                <div className={`${card} overflow-hidden`}>
                    <div className="flex flex-wrap items-center gap-3 p-5">
                        <h3 className="mr-auto text-lg font-extrabold">Daftar peserta <span className="text-sm font-semibold text-slate-500">({list.length})</span></h3>
                        <div className="relative">
                            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
                            <input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Cari ID, nama, pembeli, kode" className={`${field} w-64 pl-9`} />
                        </div>
                        <select value={type} onChange={(e) => setType(e.target.value)} className={field}>
                            <option value="">Semua jenis tiket</option>
                            {types.map((t) => <option key={t} value={t}>{t}</option>)}
                        </select>
                        <select value={status} onChange={(e) => setStatus(e.target.value)} className={field}>
                            <option value="">Semua status</option>
                            {statuses.map((s) => <option key={s} value={s}>{STATUS[s]?.label ?? s}</option>)}
                        </select>
                    </div>
                    {list.length === 0 ? (
                        <p className="border-t border-slate-200 px-5 py-12 text-center text-sm font-semibold text-slate-500 dark:border-white/10">{rows.length === 0 ? 'Belum ada pembelian tiket.' : 'Tidak ada data yang cocok.'}</p>
                    ) : (

                                        <th className="w-[210px] px-5 py-3">ID registrasi</th>
                                        <th className="px-3 py-3">Peserta</th>
                                        <th className="px-3 py-3">Jenis</th>
                                        <th className="px-3 py-3">Pembeli</th>
                                        <th className="px-3 py-3">Status</th>
                                    </tr>
                                </thead>
                                <tbody className="divide-y divide-slate-100 dark:divide-white/10">
                                    {list.map((r, i) => {
                                        const st = STATUS[r.status] ?? { label: r.status, style: 'bg-slate-200 text-slate-700' };
                                        return (
                                            <tr key={`${r.order}-${i}`} className="align-top">
                                                <td className="px-5 py-3.5 font-mono text-[13px] font-bold text-indigo-700 dark:text-indigo-300">{r.reg_id ?? <span className="font-sans font-semibold text-slate-400">-</span>}</td>
                                                <td className="px-3 py-3.5">
                                                    <div className="font-extrabold">{r.name}</div>
                                                    <div className="text-xs text-slate-500">{r.code ?? `Pesanan ${r.order}`}</div>
                                                    {r.contact && <div className="text-xs text-slate-500">{r.contact}</div>}
                                                    {r.support && <div className="text-xs font-semibold text-indigo-700">Mendukung: {r.support}</div>}
                                                </td>
                                                <td className="px-3 py-3.5"><span className="whitespace-nowrap rounded-full bg-indigo-50 px-2.5 py-1 text-xs font-bold text-indigo-700 dark:bg-white/10 dark:text-indigo-200">{r.type}</span></td>
