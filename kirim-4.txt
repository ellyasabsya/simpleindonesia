function cost(p: Pkg, subCount: number, addon: boolean, judge: boolean) {
    const base = p.payment_type === 'upfront' ? (p.price ?? 0) : 0;
    const extra = subCount === 0 ? 0 : Math.max(0, subCount - freeEvents(p));
    const sub = p.extra_sub_event_price ? extra * p.extra_sub_event_price : 0;
    const ad = addon && canAddon(p) ? (p.assessment_addon_price ?? 0) : 0;
    const aj = judge && canJudge(p) ? (p.judge_addon_price ?? 0) : 0;
    return { base, extra, sub, ad, aj, total: base + sub + ad + aj };
}

function PosterInput({ file, onChange, error }: { file: File | null; onChange: (f: File | null) => void; error?: string }) {
    const [url, setUrl] = useState<string | null>(null);
    useEffect(() => {
        if (!file) { setUrl(null); return; }
        const u = URL.createObjectURL(file);
        setUrl(u);
        return () => URL.revokeObjectURL(u);
    }, [file]);
    return (
        <div>
            <label className={label}>Poster (4:5, maks. 4 MB)</label>
            <div className="flex h-[180px] w-36 items-center justify-center overflow-hidden rounded-xl bg-gradient-to-b from-indigo-900 via-blue-700 to-purple-600 text-xs text-white/70">
                {url ? <img src={url} alt="" className="h-full w-full object-cover" /> : '4:5'}
            </div>
            <input type="file" accept="image/jpeg,image/png,image/webp" onChange={(e) => onChange(e.target.files?.[0] ?? null)}
                className="mt-3 block w-full text-xs file:mr-2 file:rounded-lg file:border-0 file:bg-indigo-50 file:px-3 file:py-2 file:text-xs file:font-bold file:text-indigo-800 dark:file:bg-indigo-500/20 dark:file:text-white" />
            <Err m={error} />
        </div>
    );
}

function JuknisInput({ file, onChange, error }: { file: File | null; onChange: (f: File | null) => void; error?: string }) {
    const [localError, setLocalError] = useState<string | null>(null);
    const pick = (f: File | null) => {
        if (f && f.size > 5 * 1024 * 1024) {
            setLocalError('Ukuran file maksimal 5 MB.');
            onChange(null);
            return;
        }
        setLocalError(null);
        onChange(f);
    };
    const size = file ? (file.size >= 1048576 ? `${(file.size / 1048576).toFixed(1)} MB` : `${Math.max(1, Math.round(file.size / 1024))} KB`) : '';
    return (
        <div>
            <label className={label}>Juknis (opsional)</label>
            <label className="flex cursor-pointer items-center gap-4 rounded-xl border-2 border-dashed border-slate-300 bg-white/70 p-4 transition hover:border-indigo-400 hover:bg-indigo-50/60 dark:border-white/15 dark:bg-white/5 dark:hover:bg-white/10">
                <input type="file" accept="application/pdf" className="hidden"
                    onChange={(e) => { pick(e.target.files?.[0] ?? null); e.target.value = ''; }} />
                <span className="flex h-11 w-11 shrink-0 items-center justify-center rounded-lg bg-indigo-100 text-indigo-800 dark:bg-indigo-500/20 dark:text-white">
                    <svg viewBox="0 0 24 24" className="h-5 w-5" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
                        <path d="M14 2H6a2 2 0 0 0-2 2v16a2 2 0 0 0 2 2h12a2 2 0 0 0 2-2V8z" />
                        <path d="M14 2v6h6" />
                        <path d="M9 13h6M9 17h6" />
                    </svg>
                </span>
                <span className="min-w-0 flex-1">
                    <span className="block truncate text-sm font-bold">{file ? file.name : 'Pilih file PDF juknis'}</span>
                    <span className="block text-xs text-slate-500 dark:text-slate-400">{file ? size : 'Petunjuk teknis · PDF · maks. 5 MB'}</span>
                </span>
                {file ? (
                    <button type="button" onClick={(e) => { e.preventDefault(); onChange(null); }}
                        className="rounded-lg border border-rose-200 bg-rose-50 px-3 py-2 text-xs font-bold text-rose-700 hover:bg-rose-100 dark:border-rose-500/30 dark:bg-rose-500/15 dark:text-rose-200">
                        Hapus
                    </button>
                ) : (
                    <span className="rounded-lg bg-indigo-50 px-3 py-2 text-xs font-bold text-indigo-800 dark:bg-indigo-500/20 dark:text-white">Pilih File</span>
                )}
            </label>
            <Err m={localError ?? error} />
        </div>
    );
}

    const unit = isTeam ? 'tim' : 'orang';
    const teamN = Math.min(parseInt(d.team_size, 10) || 0, 100);
    const perRow = Math.max(parseInt(d.team_per_row, 10) || 1, 1);
    const leaderN = Math.min(parseInt(d.team_leaders, 10) || 0, teamN);
    const memberN = Math.max(teamN - leaderN, 0);
    const buildSlots = (n: number, l: number) => Array.from({ length: n }, (_, i) => (i < l ? (l > 1 ? `Ketua ${i + 1}` : 'Ketua') : `Peserta ${i - l + 1}`));
    const updateTeam = (p: { size?: string; row?: string; lead?: string }) => {
        const size = p.size ?? d.team_size;
        const row = p.row ?? d.team_per_row;
        const lead = p.lead ?? d.team_leaders;
        const n = Math.min(parseInt(size, 10) || 0, 100);
        const l = Math.min(parseInt(lead, 10) || 0, n);
        set({ team_size: n ? String(n) : '', team_per_row: row, team_leaders: lead, team_slots: buildSlots(n, l) });
    };
    const add = (v: string) => {
        const name = v.trim();
        if (!name || full || idx(name) >= 0) return;
        set({ participant_categories: [...list, newCat(name)] });
    };
    const remove = (i: number) => set({ participant_categories: list.filter((_, j) => j !== i) });
    const setQuota = (i: number, v: string) => set({ participant_categories: list.map((c, j) => (j === i ? { ...c, quota: v.replace(/\D/g, '') } : c)) });
    const patch = (i: number, p: Partial<(typeof list)[number]>) => set({ participant_categories: list.map((c, j) => (j === i ? { ...c, ...p } : c)) });
    const toggle = (v: string) => { const i = idx(v); if (i >= 0) remove(i); else add(v); };
    const options = [...PRESET_CATEGORIES, ...list.map((x) => x.name).filter((n) => !PRESET_CATEGORIES.some((p) => p.toLowerCase() === n.toLowerCase()))];
    return (
        <div className="space-y-5">
            {comp && (
                <div className={`${box} space-y-5 p-5`}>
                    <div>
                        <b className="block text-[15px]">Jenis perlombaan *</b>
                        <p className={help}>Pilih apakah peserta mendaftar sendiri atau sebagai tim.</p>
                    </div>
                    <div className="grid gap-3 sm:grid-cols-2">
                        {[['individual', 'Perorangan', 'Peserta mendaftar sendiri-sendiri.'], ...(canTeam ? [['team', 'Tim', 'Peserta mendaftar sebagai satu tim.']] : [])].map(([v, t, sub]) => (
                            <label key={v} className={`flex cursor-pointer items-start gap-3 rounded-2xl border p-4 ${d.competition_type === v ? 'border-indigo-400 bg-indigo-50 dark:bg-indigo-500/20' : 'border-slate-200 bg-white/70 dark:border-white/10 dark:bg-white/5'}`}>
                                <input type="radio" className="mt-1 accent-indigo-700" checked={d.competition_type === v} onChange={() => set({ competition_type: v })} />
                                <span><b className="block text-sm">{t}</b><span className="text-xs text-slate-500 dark:text-slate-400">{sub}</span></span>
                            </label>
                        ))}
                    </div>
                    <Err m={err(pre + 'competition_type')} />
                    {!canTeam && <p className={help}>Perlombaan tim hanya tersedia di paket dengan fitur Manajemen Tim.</p>}

                    {isTeam && (
                        <div className="space-y-4 border-t border-slate-200/70 pt-4 dark:border-white/10">
                            <div className="grid gap-4 sm:grid-cols-3">
                                {([['team_size', 'Jumlah peserta per tim', 'Contoh: 6'], ['team_per_row', 'Peserta per baris', 'Contoh: 3'], ['team_leaders', 'Kotak ketua', 'Contoh: 1']] as const).map(([k, t, ph]) => (
                                    <div key={k}>
                                        <label className={label}>{t} *</label>
                                        <input inputMode="numeric" className="h-11 w-full rounded-xl border border-slate-200 bg-white/90 px-3 text-[15px] outline-none focus:border-indigo-400 dark:border-white/10 dark:bg-slate-800"
                                            value={d[k]} placeholder={ph}
                                            onChange={(ev) => { const v = ev.target.value.replace(/\D/g, ''); updateTeam(k === 'team_size' ? { size: v } : k === 'team_per_row' ? { row: v } : { lead: v }); }} />
                                        <Err m={err(pre + k)} />
                                    </div>
                                ))}
                            </div>
                            <p className={help}>Jumlah peserta per tim sudah termasuk kotak ketua.</p>
                            <div className="rounded-2xl border border-slate-200 bg-white/70 p-5 text-center dark:border-white/10 dark:bg-white/5">
                                <b className="block text-sm">Preview Barisan</b>
                                {teamN > 0 ? (
                                    <>
                                        <div className="my-4 flex items-start justify-center gap-6">
                                            <div className="grid gap-2" style={{ gridTemplateColumns: `repeat(${perRow}, 2.75rem)` }}>
                                                {Array.from({ length: memberN }, (_, n) => (
                                                    <div key={n} title={`Peserta ${n + 1}`} className="h-11 w-11 rounded-xl bg-sky-400" />
                                                ))}
                                            </div>
                                            {leaderN > 0 && (
                                                <div className="flex flex-wrap gap-2">
                                                    {Array.from({ length: leaderN }, (_, n) => (
                                                        <div key={n} title="Ketua" className="h-11 w-11 rounded-xl bg-rose-500" />
                                                    ))}
                                                </div>
                                            )}
                                        </div>
                                        <p className="text-xs font-medium text-slate-600 dark:text-slate-300">{memberN} peserta (biru) · {leaderN} ketua (merah) · {perRow} per baris</p>
                                    </>
                                ) : (
                                    <p className={`${help} mt-3`}>Isi jumlah peserta per tim untuk melihat susunan barisan.</p>
                                )}
                            </div>
                        </div>
                    )}
                </div>
            )}

            <div className={`${box} space-y-4 p-5`}>
                <div>
                    <b className="block text-[15px]">Kategori peserta *</b>
                    <p className={help}>
                        {max <= 10 ? `Paket ini maksimal ${max} kategori (${list.length} dipilih).` : 'Pilih kategori yang sesuai. Setiap kategori punya kuota sendiri dan jumlahnya tidak dibatasi.'}
                    </p>
                </div>

                <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-3 lg:grid-cols-6">
                    {options.map((o) => {
                        const on = idx(o) >= 0;
                        return (
                            <button key={o} type="button" onClick={() => toggle(o)} disabled={!on && full}
                                className={`flex h-11 items-center justify-center gap-1.5 rounded-xl border px-3 text-sm font-semibold transition disabled:cursor-not-allowed disabled:opacity-40 ${on ? 'border-indigo-500 bg-indigo-600 text-white shadow-sm' : 'border-slate-200 bg-white/80 text-slate-700 hover:border-indigo-300 dark:border-white/10 dark:bg-white/5 dark:text-slate-200'}`}>
                                {on && <Check className="h-4 w-4 shrink-0" />}
                                <span className="truncate">{o}</span>
                            </button>
                        );
                    })}
                </div>

                <div className="flex gap-2 border-t border-slate-200/70 pt-4 dark:border-white/10">
                    <input className="h-11 min-w-0 flex-1 rounded-xl border border-slate-200 bg-white/90 px-3.5 text-sm outline-none focus:border-indigo-400 dark:border-white/10 dark:bg-slate-800"
                        value={draft} onChange={(e) => setDraft(e.target.value)} maxLength={60} disabled={full}
                        onKeyDown={(e) => { if (e.key === 'Enter') { e.preventDefault(); add(draft); setDraft(''); } }}
                        placeholder="Tidak ada di daftar? Ketik nama kategori lalu tekan Tambah" />
                    <button type="button" onClick={() => { add(draft); setDraft(''); }} disabled={full}
                        className="h-11 shrink-0 rounded-xl bg-slate-800 px-6 text-sm font-bold text-white transition hover:bg-slate-700 disabled:opacity-40 dark:bg-white/20">Tambah</button>
                </div>
                {list.length === 0 && <Err m={err(pre + 'participant_categories')} />}
            </div>

            {list.length > 0 && (
                <div className={`${box} divide-y divide-slate-200/70 dark:divide-white/10`}>
                    {list.map((c, i) => (
                        <div key={c.name} className="flex flex-wrap items-center gap-3 px-4 py-3">
                            <span className="min-w-[6rem] flex-1 font-bold">{c.name}</span>
                            <div>
                                <div className="flex items-center gap-2">
                                    <input inputMode="numeric" className="h-11 w-32 rounded-xl border border-slate-200 bg-white/90 px-3 text-right text-[15px] outline-none focus:border-indigo-400 dark:border-white/10 dark:bg-slate-800"
                                        value={c.quota} onChange={(e) => setQuota(i, e.target.value)} placeholder="Kuota" />
                                    <span className="text-sm text-slate-500 dark:text-slate-400">{unit}</span>
                                </div>
                                <Err m={err(`${pre}participant_categories.${i}.quota`)} />
                            </div>
                            <button type="button" onClick={() => remove(i)} className="text-sm font-bold text-rose-500">Hapus</button>
                            {comp && <CompFields c={c} i={i} patch={patch} err={err} pre={pre} kind={d.category === 'olahraga' ? 'pertandingan' : 'perlombaan'} />}
                        </div>
                    ))}
                    <div className="flex items-center justify-between px-4 py-3 font-extrabold">
                        <span>Total kuota event</span>
                        <span className="text-indigo-700 dark:text-indigo-300">{totalQuota(d).toLocaleString('id-ID')} {unit}</span>
                    </div>
                </div>
            )}

            {ticketOnly ? (
                <div className={`${box} p-4 text-sm`}><b className="block">Pendaftaran peserta: Tiket</b><span className="text-xs text-slate-500 dark:text-slate-400">Paket ini otomatis mencatat pendaftaran lewat tiket, tanpa transfer manual.</span></div>
            ) : (
            <div>
                <>
                <div className={`${box} mb-6 p-5`}>
                    <b className="block text-[15px]">Biaya pendaftaran</b>
                    <span className="mt-1 block text-xs text-slate-500 dark:text-slate-400">Biaya per peserta. Kosongkan atau isi 0 kalau pendaftaran gratis.</span>
                    <div className="mt-3 flex items-center overflow-hidden rounded-2xl border border-slate-200 bg-white focus-within:border-indigo-500 focus-within:ring-4 focus-within:ring-indigo-500/15 dark:border-white/10 dark:bg-slate-950/60">
                        <span className="border-r border-slate-200 px-4 py-3 text-sm font-bold text-slate-500 dark:border-white/10 dark:text-slate-400">Rp</span>
                        <input
                            inputMode="numeric"
                            value={d.registration_fee ? new Intl.NumberFormat('id-ID').format(Number(d.registration_fee)) : ''}
                            onChange={(e) => set({ registration_fee: e.target.value.replace(/\D/g, '').slice(0, 9) })}
                            placeholder="0"
                            className="w-full bg-transparent px-4 py-3 text-sm outline-none dark:text-white"
                        />
                    </div>
                    <Err m={err(pre + 'registration_fee')} />
                </div>
                <label className={label}>Pendaftaran peserta *</label></>
                <div className="grid gap-3 sm:grid-cols-2">
                    {[['transfer_manual', 'Transfer Manual', 'Peserta transfer, panitia memverifikasi pembayaran.'], ...(canTicket ? [['ticket', 'Tiket', 'Peserta membeli tiket lewat platform.']] : [])].map(([v, t, sub]) => (
                        <label key={v} className={`flex cursor-pointer items-start gap-3 rounded-2xl border p-4 ${d.registration_method === v ? 'border-indigo-400 bg-indigo-50 dark:bg-indigo-500/20' : 'border-slate-200 bg-white/70 dark:border-white/10 dark:bg-white/5'}`}>
                            <input type="radio" className="mt-1 accent-indigo-700" checked={d.registration_method === v} onChange={() => set({ registration_method: v, ticket_types: v === 'ticket' && d.ticket_types.length === 0 ? [newTicket()] : d.ticket_types })} />
                            <span><b className="block text-sm">{t}</b><span className="text-xs text-slate-500 dark:text-slate-400">{sub}</span></span>
                        </label>
                    ))}
                </div>
                <Err m={err(pre + 'registration_method')} />
                {!canTicket && <p className={help}>Pendaftaran lewat tiket hanya tersedia di paket dengan fitur Tiket.</p>}
            </div>
            )}

            {d.registration_method === 'ticket' && (
                <div className="space-y-4 rounded-2xl border border-indigo-200 bg-indigo-50/60 p-5 dark:border-white/10 dark:bg-indigo-500/10">
                    <div>
                        <h4 className="text-sm font-bold text-slate-900 dark:text-white">Aktifkan Pendaftaran Tiket <span className="text-red-500">*</span></h4>
                        <p className="text-xs text-slate-600 dark:text-slate-300">Wajib diisi. Tanpa jenis tiket, pembeli tidak bisa mendaftar.</p>
                    </div>
                    {d.ticket_types.map((t, i) => {
                        const up = (patch: Partial<TicketType>) => set({ ticket_types: d.ticket_types.map((x, j) => (j === i ? { ...x, ...patch } : x)) });
                        const k = pre + `ticket_types.${i}.`;
                        const inp = 'w-full rounded-xl border border-slate-200 bg-white px-3 py-2.5 text-sm dark:border-white/10 dark:bg-white/5';
                        return (
                            <div key={i} className="space-y-3 rounded-xl border border-slate-200 bg-white/80 p-4 dark:border-white/10 dark:bg-white/5">
                                <div className="flex items-center justify-between">
                                    <span className="text-xs font-bold text-slate-500">Jenis tiket #{i + 1}</span>
                                    {d.ticket_types.length > 1 && (
                                        <button type="button" onClick={() => set({ ticket_types: d.ticket_types.filter((_, j) => j !== i) })} className="text-xs font-bold text-red-600">Hapus</button>
                                    )}
                                </div>
                                <div className="grid gap-3 sm:grid-cols-2">
                                    <div>
                                        <label className="mb-1 block text-xs font-bold">Nama tiket *</label>
                                        <input className={inp} value={t.name} maxLength={60} placeholder="Contoh: Reguler, VIP" onChange={(e) => up({ name: e.target.value })} />
                                        <Err m={err(k + 'name')} />
                                    </div>
                                    <div>
                                        <label className="mb-1 block text-xs font-bold">Harga (Rp) *</label>
                                        <input className={inp} inputMode="numeric" placeholder="0 jika gratis"
                                            value={t.price ? new Intl.NumberFormat('id-ID').format(Number(t.price)) : t.price}
                                            onChange={(e) => up({ price: e.target.value.replace(/\D/g, '').slice(0, 9) })} />
                                        <Err m={err(k + 'price')} />
                                    </div>
                                    <div>
                                        <label className="mb-1 block text-xs font-bold">Kuota tiket *</label>
                                        <input className={inp} inputMode="numeric" value={t.quota} onChange={(e) => up({ quota: e.target.value.replace(/\D/g, '').slice(0, 7) })} />
                                        <Err m={err(k + 'quota')} />
                                    </div>
                                    <div>
                                        <label className="mb-1 block text-xs font-bold">Maksimal per pesanan (1-10) *</label>
                                        <input className={inp} inputMode="numeric" value={t.max_per_order} onChange={(e) => up({ max_per_order: e.target.value.replace(/\D/g, '').slice(0, 2) })} />
                                        <Err m={err(k + 'max_per_order')} />
                                    </div>
                                    <div>
                                        <label className="mb-1 block text-xs font-bold">Berlaku mulai *</label>
                                        <input type="date" className={inp} value={t.valid_from} onChange={(e) => up({ valid_from: e.target.value })} />
                                        <Err m={err(k + 'valid_from')} />
                                    </div>
                                    <div>
                                        <label className="mb-1 block text-xs font-bold">Berlaku sampai *</label>
                                        <input type="date" className={inp} value={t.valid_until} onChange={(e) => up({ valid_until: e.target.value })} />
                                        <Err m={err(k + 'valid_until')} />
                                    </div>
                                </div>
                            </div>
                        );
                    })}
                    <Err m={err(pre + 'ticket_types')} />
                    {d.ticket_types.length < 10 && (
                        <button type="button" onClick={() => set({ ticket_types: [...d.ticket_types, newTicket()] })} className="rounded-xl border border-indigo-300 px-4 py-2 text-sm font-bold text-indigo-700 dark:text-indigo-200">+ Tambah jenis tiket</button>
                    )}
                </div>
            )}

            {d.registration_method === 'transfer_manual' && (
                <div className={`${box} space-y-4 p-4`}>
                    <div>
                        <b className="block text-sm">Rekening tujuan pembayaran</b>
                        <p className={help}>Peserta akan diminta transfer ke rekening ini, lalu panitia memverifikasi pembayarannya.</p>
                    </div>
                    <div className="grid gap-4 sm:grid-cols-3">
                        {([['bank_name', 'Nama bank', 'Contoh: BCA, Mandiri, BRI'], ['bank_account_number', 'Nomor rekening', 'Hanya angka'], ['bank_account_name', 'Atas nama', 'Nama pemilik rekening']] as const).map(([k, t, ph]) => (
                            <div key={k}>
                                <label className={label}>{t} *</label>
                                <input className="h-11 w-full rounded-xl border border-slate-200 bg-white/90 px-3 text-[15px] outline-none focus:border-indigo-400 dark:border-white/10 dark:bg-slate-800"
                                    inputMode={k === 'bank_account_number' ? 'numeric' : 'text'} maxLength={k === 'bank_account_number' ? 40 : 100}
                                    value={d[k]} placeholder={ph}
                                    onChange={(ev) => set({ [k]: k === 'bank_account_number' ? ev.target.value.replace(/[^0-9 -]/g, '') : ev.target.value })} />
                                <Err m={err(pre + k)} />
                            </div>
                        ))}
                    </div>
                </div>
            )}
        </div>
    );
}

function PosterThumb({ file }: { file: File | null }) {
    const [url, setUrl] = useState<string | null>(null);
    useEffect(() => {
        if (!file) { setUrl(null); return; }
        const u = URL.createObjectURL(file);
        setUrl(u);
        return () => URL.revokeObjectURL(u);
    }, [file]);
    return (
        <div className="w-full shrink-0 self-start overflow-hidden rounded-2xl bg-gradient-to-b from-indigo-900 via-blue-700 to-purple-600 shadow-md sm:w-44">
            {url ? <img src={url} alt="Poster" className="block h-auto w-full" /> : <div className="flex aspect-[4/5] items-center justify-center text-xs font-semibold text-white/70">Tanpa poster</div>}
        </div>
    );
}

                        </button>

                    {d.description && <p className="line-clamp-3 text-sm text-slate-600 dark:text-slate-300">{d.description}</p>}
                    <div className="grid gap-4 sm:grid-cols-2">
                        <Fact icon={CalendarDays} title="Tanggal">{fmt(d.starts_at)} – {fmt(d.ends_at)}</Fact>
                        {!main && <Fact icon={MapPin} title="Lokasi">{d.location || '-'}</Fact>}
                        {!main && <Fact icon={Tag} title="Kategori event">{categories[d.category] ?? '-'}</Fact>}

                    {!main && d.participant_categories.length > 0 && (

                            <p className="mb-1.5 text-xs text-slate-500 dark:text-slate-400">Kategori peserta</p>
                            <div className="flex flex-wrap gap-2">
                                {d.participant_categories.map((c) => <span key={c.name} className="rounded-full bg-slate-100 px-3 py-1 text-xs font-semibold dark:bg-white/10">{c.competition_name ? `${c.competition_name} (${c.name})` : c.name} · {c.quota}</span>)}
                            </div>
                        </div>
                    )}
                </div>
            </div>
        </div>
    );
}

export default function CreateEvent({ packages, categories, selected_package_id, payment_methods = [], platform_bank = null }: Props & { payment_methods?: string[]; platform_bank?: { bank_name: string; bank_account_number: string; bank_account_holder: string } | null }) {
    const initial = packages.find((p) => p.id === selected_package_id) ?? null;
    const [step, setStep] = useState(initial && !hasOptions(initial) ? 1 : 0);
    const [local, setLocal] = useState<Errs>({});

    const { data, setData, post, processing, errors, transform } = useForm<Detail & { package_id: number | ''; assessment_addon: boolean; judge_addon: boolean; payment_method: string; payment_proof: File | null; subs: Detail[] }>({
        ...blank(),
        package_id: initial?.id ?? '',
        assessment_addon: false,
        judge_addon: false,
        payment_method: '',
        payment_proof: null,
        subs: initialSubs(initial),
    });
    const payMethod = payment_methods.length === 1 ? payment_methods[0] : data.payment_method;

    const pkg = packages.find((p) => p.id === data.package_id);
    const series = !!pkg && (isSeriesPkg(pkg) || data.subs.length > 0);
    const keys = series
        ? ['paket', 'utama', ...data.subs.map((_, i) => `sub:${i}`), 'tinjau']
        : ['paket', 'info', 'peserta', 'pj', 'tinjau'];
    const titleOf = (k: string) =>
        k.startsWith('sub:') ? `Event ${Number(k.slice(4)) + 1}` :
        ({ paket: 'Paket', info: 'Informasi Event', peserta: 'Peserta & Pendaftaran', pj: 'Penanggung Jawab', utama: 'Event Utama', tinjau: 'Tinjau' } as Record<string, string>)[k];
    const cur = keys[Math.min(step, keys.length - 1)];
    const last = step >= keys.length - 1;
    useEffect(() => {
        if (pkg?.payment_type !== 'revenue_share') return;
        if (data.registration_method !== 'ticket' || data.subs.some((x) => x.registration_method !== 'ticket')) {
            setData((d) => ({ ...d, registration_method: 'ticket', subs: d.subs.map((x) => ({ ...x, registration_method: 'ticket' })) }));
        }
    }, [pkg?.id, data.subs.length, data.registration_method]);
    const c = pkg ? cost(pkg, data.subs.length, data.assessment_addon, data.judge_addon) : null;
    const cap = maxCats(pkg);
    const bounds: Bounds = { min: data.starts_at, max: data.ends_at };

    const all: Errs = { ...(errors as Errs), ...local };
    const err = (k: string) => all[k] ?? all[Object.keys(all).find((x) => x.startsWith(k + '.')) ?? ''];

    const setMain = (patch: Partial<Detail>) => setData((d) => ({ ...d, ...patch }));
    const setSub = (i: number, patch: Partial<Detail>) => setData('subs', data.subs.map((s, j) => (j === i ? { ...s, ...patch } : s)));

    const choosePackage = (p: Pkg) => setData((d) => ({
        ...d,
        package_id: p.id,
        subs: initialSubs(p),
        assessment_addon: false,
        judge_addon: false,
        participant_categories: d.participant_categories.slice(0, maxCats(p)),
        registration_method: d.registration_method === 'ticket' && !p.can_ticket ? '' : d.registration_method,
        competition_type: d.competition_type === 'team' && !p.can_team ? '' : d.competition_type,
    }));
    // Isian = jumlah event TAMBAHAN di luar yang sudah termasuk paket.
    const setExtra = (raw: number) => {
        if (!pkg) return;
        const e = Math.max(0, Math.min(49, Number.isFinite(raw) ? raw : 0));
        const total = freeEvents(pkg) + e;
        const n = isSeriesPkg(pkg) ? total : e === 0 ? 0 : total;
        if (n < data.subs.length) {
            const lost = data.subs.slice(n).some((x) => x.title.trim());
            if (lost && !confirm('Event yang sudah terisi akan dihapus. Lanjutkan?')) return;
            setData('subs', data.subs.slice(0, n));
        } else {
            setData('subs', [...data.subs, ...blanks(n - data.subs.length)]);
        }
    };

    const validate = (key: string) => {
        let e: Errs = {};
        if (key === 'paket') { if (!data.package_id) e.package_id = 'Pilih paket event.'; }
        else if (key === 'utama') e = vUtama(data);
        else if (key === 'info') e = vInfo(data, '');
        else if (key === 'peserta') e = vPeserta(data, '');
        else if (key === 'pj') e = vPj(data, '');
        else if (key.startsWith('sub:')) {
            const i = Number(key.slice(4));
            const s = data.subs[i];
            const pre = `subs.${i}.`;
            e = { ...vInfo(s, pre, bounds), ...vPeserta(s, pre), ...vPj(s, pre) };
        }
        setLocal(e);
        return Object.keys(e).length === 0;
    };

    const next = () => {
        if (!validate(cur)) return;
        if (cur === 'utama') {
            setData('subs', data.subs.map((s) => ({ ...s, starts_at: s.starts_at || data.starts_at, ends_at: s.ends_at || data.ends_at })));
        }
        setStep(step + 1);
    };
    const back = () => { setLocal({}); setStep(step - 1); };

    const stepOfField = (k: string) => {
        if (k === 'subs' || k === 'package_id' || k === 'assessment_addon' || k === 'judge_addon') return 'paket';
        const m = k.match(/^subs\.(\d+)\./);
        if (m) return `sub:${m[1]}`;
        if (series) return ['title', 'description', 'starts_at', 'ends_at', 'poster', 'juknis'].includes(k) ? 'utama' : 'tinjau';
        if (['title', 'category', 'location', 'starts_at', 'ends_at', 'description', 'poster', 'juknis'].includes(k)) return 'info';
        if (k.startsWith('participant_') || k === 'registration_method' || k === 'registration_fee' || k.includes('ticket_types')) return 'peserta';
        if (k.startsWith('contact_')) return 'pj';
        return 'tinjau';
    };
    const submit = () => {
        setLocal({});
        transform((d) => ({ ...d, payment_method: payMethod }));
        post('/panel/ajukan', {
            forceFormData: true,
            onError: (errs) => {
                const idx = Math.min(...Object.keys(errs).map((k) => { const i = keys.indexOf(stepOfField(k)); return i < 0 ? keys.length - 1 : i; }));
                setStep(idx);
            },
        });
    };

    const h2 = 'text-2xl font-extrabold';
    const h3 = 'mb-4 text-lg font-extrabold';

    return (
        <UserLayout title="Ajukan Event" hasPanel>
            <section className="flex items-center justify-between gap-4">
                <h1 className="text-3xl font-extrabold">
                    Ajukan <span className="title-gradient">{series ? 'Rangkaian Event' : 'Event Baru'}</span>
                </h1>
                <Link href="/panel" className="flex items-center gap-2 rounded-2xl bg-white/80 px-4 py-2.5 text-sm font-bold shadow-sm hover:bg-white dark:bg-white/10">
                    <ArrowLeft className="h-4 w-4" /> Panel Lembaga
                </Link>
            </section>

            <ol className="flex items-center py-1">
                {keys.map((k, i) => (
                    <li key={k} className="flex flex-1 items-center last:flex-none">
                        <span className="flex items-center gap-3">
                            <span className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-full text-sm font-extrabold transition ${i < step ? 'bg-emerald-500 text-white' : i === step ? 'bg-gradient-to-br from-blue-600 to-violet-600 text-white shadow-lg shadow-indigo-600/30' : 'bg-white/80 text-slate-400 dark:bg-white/10'}`}>
                                {i < step ? <Check className="h-5 w-5" /> : i + 1}
                            </span>
                            <span className={`hidden text-sm font-bold lg:block ${i === step ? '' : 'text-slate-400'}`}>{titleOf(k)}</span>
                        </span>
                        {i < keys.length - 1 && <span className={`mx-3 h-1 flex-1 rounded-full ${i < step ? 'bg-emerald-500' : 'bg-slate-200 dark:bg-white/10'}`} />}
                    </li>
                ))}
            </ol>

            <div className={`${card} w-full`} key={cur}>
                {cur === 'paket' && (
                    <div className="space-y-5">
                        <div>
                            <h2 className={h2}>Pilih paket</h2>
                            <p className="mt-1 text-sm text-slate-500 dark:text-slate-400">Paket menentukan fitur dan biaya event Anda. Fitur dan bagi hasil dikunci saat event diajukan.</p>
                        </div>
                        {packages.length === 0 && <p className="text-sm text-slate-500">Belum ada paket tersedia. Hubungi admin.</p>}
                        {/* pkg-card-v2 */}
                        <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-3">
                            {packages.map((p) => {
                                const on = data.package_id === p.id;
                                const m = p.price_label.match(/^(Bayar di muka|Bagi hasil)\s+(.*)$/i);
                                const big = m ? m[2] : p.price_label;
                                const small = m ? m[1] : '';
                                const kind = isSeriesPkg(p) ? 'Rangkaian event' : p.extra_sub_event_price !== null ? 'Tunggal / rangkaian' : 'Event tunggal';
                                return (
                                    <label key={p.id} className={`relative flex h-full cursor-pointer flex-col rounded-3xl border-2 p-5 transition ${on ? 'border-indigo-500 bg-indigo-50 shadow-lg shadow-indigo-500/10 dark:bg-indigo-500/15' : 'border-slate-200 bg-white/80 hover:-translate-y-0.5 hover:border-indigo-300 hover:shadow-md dark:border-white/10 dark:bg-white/5'}`}>
                                        <input type="radio" name="package" checked={on} onChange={() => choosePackage(p)} className="sr-only" />
                                        <span className={`absolute right-4 top-4 flex h-7 w-7 items-center justify-center rounded-full border-2 transition ${on ? 'border-indigo-600 bg-indigo-600 text-white' : 'border-slate-300 text-transparent dark:border-white/20'}`}>
                                            <Check className="h-4 w-4" />
                                        </span>
                                        <span className="w-fit rounded-full bg-slate-100 px-3 py-1 text-[11px] font-bold text-slate-600 dark:bg-white/10 dark:text-slate-300">{kind}</span>
                                        <span className="mt-3 text-xl font-extrabold">{p.name}</span>
                                        <span className="mt-3 block min-h-[3.25rem]">
                                            {small && <span className="block text-xs font-semibold text-slate-500 dark:text-slate-400">{small}</span>}
                                            <span className="block text-3xl font-extrabold leading-tight text-indigo-700 dark:text-indigo-300">{big}</span>
                                        </span>
                                        {p.description && <span className="mt-1 block text-sm text-slate-500 dark:text-slate-400">{p.description}</span>}
                                        <span className="my-4 h-px bg-slate-200 dark:bg-white/10" />
                                        <span className="flex flex-1 flex-col gap-2">
                                            {p.features.slice(0, 3).map((f) => (
                                                <span key={f} className="flex items-start gap-2 text-sm">
                                                    <Check className="mt-0.5 h-4 w-4 shrink-0 text-emerald-500" /> {f}
                                                </span>
                                            ))}
                                            {p.features.length > 3 && <span className="pl-6 text-xs font-semibold text-indigo-600 dark:text-indigo-300">+{p.features.length - 3} fitur lainnya</span>}
                                        </span>
                                    </label>
                                );
                            })}
                        </div>
                        <Err m={err('package_id')} />

                        {pkg && c && (
                            <div className="rounded-2xl border border-indigo-200 bg-indigo-50/60 p-5 dark:border-indigo-400/30 dark:bg-indigo-500/10">
                                <div className="flex flex-wrap items-center justify-between gap-4">
                                    <div>
                                        <p className="text-sm text-slate-500 dark:text-slate-400">Paket dipilih</p>
                                        <p className="text-2xl font-extrabold">{pkg.name}</p>
                                        <p className="text-sm font-semibold text-indigo-700 dark:text-indigo-300">
                                            {data.subs.length === 0 ? 'Event tunggal' : `Rangkaian ${data.subs.length} event`}
                                        </p>
                                    </div>
                                    <div className="text-right">
                                        <p className="text-sm text-slate-500 dark:text-slate-400">Biaya di muka</p>
                                        <p className="text-3xl font-extrabold text-indigo-700 dark:text-indigo-300">
                                            {pkg.payment_type === 'revenue_share' ? 'Rp0' : c.total === 0 ? 'Gratis' : rp(c.total)}
                                        </p>
                                        {pkg.payment_type === 'revenue_share' && <p className="text-xs text-slate-500">{pkg.price_label}</p>}
                                    </div>
                                </div>

                                {pkg.extra_sub_event_price !== null && (
                                    <div className="mt-4 flex flex-wrap items-center justify-between gap-3 border-t border-indigo-200 pt-4 dark:border-white/10">
                                        <div>
                                            <p className="font-bold">Tambah event?</p>
                                            <p className="text-xs text-slate-500 dark:text-slate-400">
                                                {freeEvents(pkg)} event sudah termasuk. Tiap tambahan {rp(pkg.extra_sub_event_price)}.
                                            </p>
                                        </div>
                                        <div className="flex items-center gap-3">
                                            <button type="button" aria-label="Kurangi" disabled={c.extra === 0}
                                                onClick={() => setExtra(c.extra - 1)}
                                                className="flex h-11 w-11 items-center justify-center rounded-xl border border-slate-300 bg-white text-xl font-bold disabled:opacity-40 dark:border-white/20 dark:bg-slate-800">−</button>
                                            <span className="w-10 text-center text-2xl font-extrabold">{c.extra}</span>
                                            <button type="button" aria-label="Tambah" disabled={c.extra >= 49}
                                                onClick={() => setExtra(c.extra + 1)}
                                                className="flex h-11 w-11 items-center justify-center rounded-xl bg-indigo-600 text-xl font-bold text-white disabled:opacity-40">+</button>
                                        </div>
                                    </div>
                                )}

                                {canAddon(pkg) && (
                                    <label className="mt-4 flex cursor-pointer items-start gap-3 border-t border-indigo-200 pt-4 text-sm dark:border-white/10">
                                        <input type="checkbox" checked={data.assessment_addon} onChange={(e) => setData('assessment_addon', e.target.checked)} className="mt-0.5 h-4 w-4 accent-indigo-700" />
                                        <span>
                                            <b>Tambah penilaian juri</b> (+{rp(pkg.assessment_addon_price ?? 0)})
                                            <span className="block text-xs text-slate-500 dark:text-slate-400">Sistem penilaian dan akun juri untuk memberi nilai pada peserta.</span>
                                        </span>
                                    </label>
                                )}

                                {canJudge(pkg) && (
                                    <label className="mt-4 flex cursor-pointer items-start gap-3 border-t border-indigo-200 pt-4 text-sm dark:border-white/10">
                                        <input type="checkbox" checked={data.judge_addon} onChange={(e) => setData('judge_addon', e.target.checked)} className="mt-0.5 h-4 w-4 accent-indigo-700" />
                                        <span>
                                            <b>Tambah juri</b> (+{rp(pkg.judge_addon_price ?? 0)})
                                            <span className="block text-xs text-slate-500 dark:text-slate-400">Akun juri untuk memberi nilai pada peserta.</span>
                                        </span>
                                    </label>
                                )}
                                <Err m={err('subs')} />
                            </div>
                        )}

                        
                    </div>
                )}

                {cur === 'utama' && (
                    <div className="space-y-4">
                        <div>
                            <h2 className={h2}>Event utama</h2>
                            <p className="mt-1 text-sm text-slate-500 dark:text-slate-400">Ini hanya judul dan periode pembungkus rangkaian, bukan event tersendiri. Kategori, lokasi, peserta, dan penanggung jawab diisi pada tiap event di langkah berikutnya.</p>
                        </div>
                        <UtamaSection d={data} set={setMain} pre="" err={err} />
                    </div>
                )}

                {cur === 'info' && (
                    <div className="space-y-4">
                        <h2 className={h2}>Informasi Event</h2>
                        <InfoSection d={data} set={setMain} pre="" err={err} categories={categories} />
                    </div>
                )}
                {cur === 'peserta' && (
                    <div className="space-y-4">
                        <h2 className={h2}>Peserta & Pendaftaran</h2>
                        <PesertaSection d={data} set={setMain} pre="" err={err} max={cap} ticketOnly={!!pkg && pkg.payment_type === 'revenue_share'} canTicket={pkg ? pkg.can_ticket : true} canTeam={pkg ? pkg.can_team : true} />
                    </div>
                )}
                {cur === 'pj' && (
                    <div className="space-y-4">
                        <h2 className={h2}>Penanggung Jawab</h2>
                        <PjSection d={data} set={setMain} pre="" err={err} />
                    </div>
                )}

                {cur.startsWith('sub:') && (() => {
                    const i = Number(cur.slice(4));
                    const s = data.subs[i];
                    const pre = `subs.${i}.`;
                    const set = (p: Partial<Detail>) => setSub(i, p);
                    return (
                        <div className="space-y-7">
                            <div>
                                <h2 className={h2}>Event {i + 1} dari {data.subs.length}</h2>
                                <p className="mt-1 flex items-center gap-2 text-sm text-slate-500 dark:text-slate-400">
                                    <CornerDownRight className="h-4 w-4" /> Bagian dari {data.title || 'event utama'} ({fmt(data.starts_at)} – {fmt(data.ends_at)})
                                    {pkg && i >= freeEvents(pkg) && pkg.extra_sub_event_price !== null && (
                                        <span className="rounded-full bg-amber-100 px-2 py-0.5 text-[11px] font-bold text-amber-700 dark:bg-amber-500/20 dark:text-amber-200">+{rp(pkg.extra_sub_event_price)}</span>
                                    )}
                                </p>
                            </div>
                            <div><h3 className={h3}>Informasi Event</h3><InfoSection d={s} set={set} pre={pre} err={err} categories={categories} bounds={bounds} /></div>
                            <div className="border-t border-slate-200/70 pt-6 dark:border-white/10"><h3 className={h3}>Peserta & Pendaftaran</h3><PesertaSection d={s} set={set} pre={pre} err={err} max={cap} ticketOnly={!!pkg && pkg.payment_type === 'revenue_share'} canTicket={pkg ? pkg.can_ticket : true} canTeam={pkg ? pkg.can_team : true} /></div>
                            <div className="border-t border-slate-200/70 pt-6 dark:border-white/10"><h3 className={h3}>Penanggung Jawab</h3><PjSection d={s} set={set} pre={pre} err={err} /></div>
                        </div>
                    );
                })()}

                {cur === 'tinjau' && pkg && c && (
                    <div className="space-y-6">
                        <h2 className={h2}>Tinjau pengajuan</h2>

                        <div className="flex flex-wrap items-center justify-between gap-4 rounded-2xl bg-gradient-to-r from-indigo-800 via-blue-600 to-violet-600 p-5 text-white shadow-lg shadow-indigo-600/20">
                            <div>
                                <p className="text-xs font-semibold uppercase tracking-wide text-white/70">Paket</p>
                                <p className="text-xl font-extrabold">{pkg.name}</p>
                                <p className="text-sm text-white/80">{pkg.price_label}</p>
                            </div>
                            <div className="text-center">
                                <p className="text-xs font-semibold uppercase tracking-wide text-white/70">Jenis</p>
                                <p className="text-lg font-extrabold">{series ? `Rangkaian ${data.subs.length} event` : 'Event tunggal'}</p>
                            </div>
                            <div className="text-right">
                                <p className="text-xs font-semibold uppercase tracking-wide text-white/70">Total di muka</p>
                                <p className="text-xl font-extrabold">{pkg.payment_type === 'revenue_share' || c.total === 0 ? 'Gratis' : rp(c.total)}</p>
                            </div>
                        </div>

                        {series ? (
                            <>
                                <ReviewCard d={data} categories={categories} badge="Event utama" main onEdit={() => setStep(keys.indexOf('utama'))} />
                                <div>
                                    <p className="mb-3 text-lg font-extrabold">Daftar event ({data.subs.length})</p>
                                    <div className="space-y-4 border-l-2 border-indigo-200 pl-5 dark:border-indigo-400/30">
                                        {data.subs.map((s, i) => (
                                            <ReviewCard key={i} d={s} categories={categories} badge={`Event ${i + 1}`} onEdit={() => setStep(keys.indexOf(`sub:${i}`))} />
                                        ))}
                                    </div>
                                </div>
                            </>
                        ) : (
                            <ReviewCard d={data} categories={categories} badge="Event" onEdit={() => setStep(keys.indexOf('info'))} />
                        )}

                        <div className="rounded-2xl border border-indigo-200 bg-indigo-50/60 p-5 text-[15px] dark:border-indigo-400/30 dark:bg-indigo-500/10">
                            <p className="mb-2 font-extrabold">Rincian biaya</p>
                            {pkg.payment_type === 'revenue_share' ? (
                                <p>Tanpa biaya di muka. {pkg.price_label} dari penjualan.</p>
                            ) : (
                                <div className="space-y-1">
                                    <div className="flex justify-between"><span>Paket {pkg.name}{pkg.included_sub_events > 0 ? ` (termasuk ${pkg.included_sub_events} event)` : ''}</span><span>{c.base === 0 ? 'Gratis' : rp(c.base)}</span></div>
                                    {c.extra > 0 && <div className="flex justify-between"><span>Event tambahan ({c.extra} × {rp(pkg.extra_sub_event_price ?? 0)})</span><span>{rp(c.sub)}</span></div>}
                                    {c.ad > 0 && <div className="flex justify-between"><span>Add-on penilaian juri</span><span>{rp(c.ad)}</span></div>}
                                    {c.aj > 0 && <div className="flex justify-between"><span>Add-on juri</span><span>{rp(c.aj)}</span></div>}
                                    <div className="mt-2 flex justify-between border-t border-indigo-200 pt-2 font-extrabold dark:border-white/10"><span>Total di muka</span><span>{c.total === 0 ? 'Gratis' : rp(c.total)}</span></div>
                                </div>
                            )}
                        </div>
                        {pkg.payment_type !== 'revenue_share' && c.total > 0 && (
                            <div className="rounded-2xl border border-slate-200 bg-white/70 p-5 text-[15px] dark:border-white/10 dark:bg-white/5">
                                <p className="mb-3 font-extrabold">Metode pembayaran</p>
                                <div className={`grid gap-3 ${payment_methods.length > 1 ? 'md:grid-cols-2' : ''}`}>
                                    {payment_methods.map((m) => {
                                        const on = payMethod === m;
                                        return (
                                            <button
                                                type="button"
                                                key={m}
                                                onClick={() => setData('payment_method', m)}
                                                className={`rounded-xl border p-4 text-left transition ${on ? 'border-indigo-400 bg-indigo-50 ring-1 ring-indigo-400/40 dark:bg-indigo-500/10' : 'border-slate-200 hover:border-indigo-300 dark:border-white/10'}`}
                                            >
                                                <b className="block">{m === 'midtrans' ? 'Midtrans (online)' : 'Transfer manual'}</b>
                                                <span className="mt-1 block text-sm text-slate-500 dark:text-slate-400">
                                                    {m === 'midtrans' ? 'Bayar online, pembayaran terkonfirmasi otomatis.' : 'Transfer ke rekening SIMPLE, pembayaran diverifikasi admin.'}
                                                </span>
                                            </button>
                                        );
                                    })}
                                </div>
                                {payMethod === 'transfer_manual' && platform_bank && (
                                    <div className="mt-4 rounded-xl bg-slate-50 p-4 text-sm dark:bg-white/5">
                                        <p className="font-bold">Rekening tujuan</p>
                                        <p>{platform_bank.bank_name} - {platform_bank.bank_account_number}</p>
                                        <p className="text-slate-500 dark:text-slate-400">a.n. {platform_bank.bank_account_holder}</p>
                                    </div>
                                )}
                                {payMethod === 'transfer_manual' && (
                                    <div className="mt-4">
                                        <label className="block text-sm font-bold">Bukti pembayaran *</label>
                                        <p className="mb-2 text-xs text-slate-500 dark:text-slate-400">Transfer sesuai total di muka, lalu unggah bukti transfer (foto otomatis dikompres, PDF maks. 2 MB).</p>
                                        <input
                                            type="file"

    private function uniqueSlug(string $title): string
    {
        $base = Str::slug($title) ?: 'event';

    /** Tiket yang sudah terpakai: lunas, atau menunggu pembayaran yang belum kedaluwarsa (sama dengan logika pembelian). */
    public static function sold(int $typeId): int
    {
        return (int) DB::table('ticket_orders')->where('ticket_type_id', $typeId)
            ->where('status', '!=', 'batal')
            ->where(fn ($q) => $q->where('status', '!=', 'menunggu')->orWhere('expires_at', '>', now()))
            ->sum('quantity');
    }

    private function guard(Request $request, Event $event): void
    {
        $inst = $request->attributes->get('institution');
        abort_unless($inst && $event->institution_id === $inst->id, 404);
        abort_unless($event->registration_method === 'ticket' && $event->status === Event::PUBLISHED, 404);
    }

    private function validated(Request $request, Event $event, ?TicketType $type = null): array
    {
        $end = $event->ends_at->toDateString();
        $data = $request->validate([
            'name' => ['required', 'string', 'max:60'],

            'support_enabled' => ['required', 'boolean'],
            'support_options' => ['nullable', 'array', 'max:50'],
            'support_options.*' => ['nullable', 'string', 'max:60'],

            'price.integer' => 'Harga harus berupa angka.',
            'quota.required' => 'Isi kuota tiket.',

            'valid_days.required' => 'Isi masa berlaku tiket.',

        ]);

        $data['name'] = trim($data['name']);

            ->unique(fn ($o) => mb_strtolower($o))->values()->all();

        }

        $dupe = TicketType::where('event_id', $event->id)

    public function toggle(Request $request, Event $event): RedirectResponse
    {
        $this->guard($request, $event);
        $data = $request->validate(['open' => ['required', 'boolean']]);

        if ($data['open'] && ! TicketType::where('event_id', $event->id)->exists()) {
            return back()->with('error', 'Tambahkan minimal satu jenis tiket dulu.');
        }

        DB::table('events')->where('id', $event->id)->update(['ticket_sales_open' => $data['open'] ? 1 : 0]);

        return back()->with('success', $data['open'] ? 'Penjualan tiket dibuka.' : 'Penjualan tiket ditutup.');
    }

    public function store(Request $request, Event $event): RedirectResponse
    {
        $this->guard($request, $event);
        $data = $this->validated($request, $event);

        TicketType::create($data + ['event_id' => $event->id]);

        return back()->with('success', 'Jenis tiket ditambahkan.');
    }

    public function update(Request $request, Event $event, TicketType $type): RedirectResponse
    {
        $this->guard($request, $event);
        abort_unless($type->event_id === $event->id, 404);
        $data = $this->validated($request, $event, $type);

        $hasOrders = DB::table('ticket_orders')->where('ticket_type_id', $type->id)->where('status', '!=', 'batal')->exists();
        if ($hasOrders && (int) $data['price'] !== (int) $type->price) {
            return back()->withErrors(['price' => 'Harga tidak bisa diubah karena sudah ada pesanan untuk jenis ini.']);
        }
        $sold = self::sold($type->id);
        if ((int) $data['quota'] < $sold) {
            return back()->withErrors(['quota' => "Kuota tidak boleh di bawah tiket yang sudah terpakai ({$sold})."]);
        }

        $type->update($data);

        return back()->with('success', 'Jenis tiket diperbarui.');
    }

    public function destroy(Request $request, Event $event, TicketType $type): RedirectResponse
    {
        $this->guard($request, $event);
        abort_unless($type->event_id === $event->id, 404);

        if (DB::table('ticket_orders')->where('ticket_type_id', $type->id)->exists()) {
            return back()->with('error', 'Jenis tiket yang sudah punya pesanan tidak bisa dihapus.');
        }
        if (TicketType::where('event_id', $event->id)->count() <= 1) {
            return back()->with('error', 'Event harus punya minimal satu jenis tiket.');
        }

        $type->delete();

        return back()->with('success', 'Jenis tiket dihapus.');
    }

    /** Ringkasan angka dashboard: induk = jumlah semua sub-event, sub-event = dirinya sendiri. */
    private function dashboardStats(Event $event): array
    {
        $db = \Illuminate\Support\Facades\DB::class;
        $isParent = $event->parent_id === null && $event->children()->exists();
        $events = $isParent ? $event->children()->get() : collect([$event]);
        $ids = $events->pluck('id')->all();
        $titles = $events->pluck('title', 'id');
        $tz = config('app.agenda_timezone', 'Asia/Jakarta');
        $closed = ['dibatalkan', 'batal', 'kedaluwarsa', 'ditolak'];
        $grp = fn ($s) => $s === 'lunas' ? 'lunas' : (in_array($s, $closed, true) ? 'batal' : 'menunggu');
        $day = fn ($d) => \Illuminate\Support\Carbon::parse($d)->setTimezone($tz)->format('Y-m-d');

        $quotas = $db::table('ticket_types')->whereIn('event_id', $ids)
            ->selectRaw('event_id, sum(quota) as q')->groupBy('event_id')->pluck('q', 'event_id');

        $perSub = [];
        foreach ($events as $e) {
            $perSub[$e->id] = [
                'id' => $e->id, 'title' => $e->title, 'method' => $e->registration_method,
                'total' => 0, 'lunas' => 0, 'menunggu' => 0, 'batal' => 0,
                'revenue' => 0, 'tickets' => 0, 'hadir' => 0, 'quota' => (int) ($quotas[$e->id] ?? 0),
            ];
        }

        $today = now($tz)->startOfDay();
        $trend = [];
        for ($i = 13; $i >= 0; $i--) {
            $trend[$today->copy()->subDays($i)->format('Y-m-d')] = ['n' => 0, 'rev' => 0];
        }

        $recent = [];

        $orders = $db::table('ticket_orders')->whereIn('event_id', $ids)
            ->get(['event_id', 'status', 'quantity', 'subtotal', 'created_at']);
        foreach ($orders as $o) {
            $g = $grp((string) $o->status);
            $q = max(1, (int) $o->quantity);
            $perSub[$o->event_id][$g] += $q;
            if ($g === 'lunas') {
                $perSub[$o->event_id]['revenue'] += (int) $o->subtotal;
            }
            if ($g !== 'batal' && $o->created_at) {
                $k = $day($o->created_at);
                if (isset($trend[$k])) {
                    $trend[$k]['n'] += $q;
                    if ($g === 'lunas') {
                        $trend[$k]['rev'] += (int) $o->subtotal;
                    }
                }
            }
        }

        $regs = $db::table('event_registrations')->whereIn('event_id', $ids)
            ->get(['event_id', 'status', 'total_fee', 'created_at']);
        foreach ($regs as $r) {
            $g = $grp((string) $r->status);
            $perSub[$r->event_id][$g] += 1;
            if ($g === 'lunas') {
                $perSub[$r->event_id]['revenue'] += (int) $r->total_fee;
            }
            if ($g !== 'batal' && $r->created_at) {
                $k = $day($r->created_at);
                if (isset($trend[$k])) {
                    $trend[$k]['n'] += 1;
                    if ($g === 'lunas') {
                        $trend[$k]['rev'] += (int) $r->total_fee;
                    }
                }
            }
        }

        $holders = $db::table('ticket_holders as h')
            ->join('ticket_orders as o', 'o.id', '=', 'h.ticket_order_id')
            ->whereIn('h.event_id', $ids)->where('o.status', 'lunas')

        foreach ($holders as $h) {

        }

        foreach ($perSub as &$p) {

        }
        unset($p);
        $sum = ['total' => 0, 'lunas' => 0, 'menunggu' => 0, 'batal' => 0, 'revenue' => 0, 'tickets' => 0, 'hadir' => 0];
        foreach ($perSub as $p) {
            foreach ($sum as $k => $_) {
                $sum[$k] += $p[$k];
            }
        }

        $latestOrders = $db::table('ticket_orders')->whereIn('event_id', $ids)->orderByDesc('created_at')->limit(8)
            ->get(['event_id', 'buyer_name', 'quantity', 'total', 'status', 'created_at']);
        foreach ($latestOrders as $o) {

                'kind' => 'tiket',
                'status' => $grp((string) $o->status),
                'note' => max(1, (int) $o->quantity) . ' tiket · Rp' . number_format((int) $o->total, 0, ',', '.'),
            ];
        }
        $latestRegs = $db::table('event_registrations')->whereIn('event_id', $ids)->orderByDesc('created_at')->limit(8)
            ->get(['event_id', 'contact_name', 'total_fee', 'status', 'created_at']);
        foreach ($latestRegs as $r) {
            $recent[] = [
                'at' => (string) $r->created_at,
                'name' => $r->contact_name ?: 'Pendaftar',
                'sub' => $titles[$r->event_id] ?? '-',
                'kind' => 'formulir',
                'status' => $grp((string) $r->status),
                'note' => (int) $r->total_fee > 0 ? 'Formulir · Rp' . number_format((int) $r->total_fee, 0, ',', '.') : 'Formulir pendaftaran',
            ];
        }
        usort($recent, fn ($a, $b) => strcmp($b['at'], $a['at']));
        $recent = array_map(function ($r) {
            $r['at'] = $r['at'] !== '' ? \Illuminate\Support\Carbon::parse($r['at'])->toIso8601String() : null;

            return $r;
        }, array_slice($recent, 0, 8));

        $trendOut = [];
        foreach ($trend as $d => $v) {
            $trendOut[] = ['date' => \Illuminate\Support\Carbon::parse($d)->format('d M'), 'n' => $v['n'], 'rev' => $v['rev']];
        }

        return [
            'isParent' => $isParent,
            'sum' => $sum,
            'subs' => array_values($perSub),
            'trend' => $trendOut,
            'recent' => $recent,
        ];
    }

    /** Progres kelengkapan tiap menu, dipakai kartu di dashboard event. */
    private function progress(Event $event): array
    {
        $db = \Illuminate\Support\Facades\DB::class;
        $base = "/panel/event/{$event->id}";
        $closed = ['dibatalkan', 'batal', 'kedaluwarsa', 'ditolak'];

        $step = fn (string $label, bool $done, string $href) => ['label' => $label, 'done' => $done, 'href' => $href];
        $make = function (array $steps): array {
            $next = collect($steps)->first(fn ($s) => ! $s['done']);

            return [
                'steps' => $steps,
                'done' => count(array_filter($steps, fn ($s) => $s['done'])),
                'total' => count($steps),
                'next' => $next ? ['label' => $next['label'], 'href' => $next['href']] : null,
            ];
        };

        $regSet = (bool) ($event->registration_start && $event->registration_end);
        $prepSet = (bool) ($event->preparation_start && $event->preparation_end);
        $agendas = $db::table('event_agendas')->where('event_id', $event->id);
        $orders = $db::table('ticket_orders')->where('event_id', $event->id);
        $types = $db::table('ticket_types')->where('event_id', $event->id);
        $vs = $db::table('voting_settings')->where('event_id', $event->id)->first();
        $parts = $db::table('voting_participants')->where('event_id', $event->id)->where('active', 1);
        $activeParts = (clone $parts)->count();

        $out = [];

        $out['kalender'] = $make([
            $step('Atur periode pendaftaran', $regSet, "$base/kalender"),
            $step('Atur periode persiapan', $prepSet, "$base/kalender"),
            $step('Tambah minimal 1 agenda', (clone $agendas)->exists(), "$base/kalender"),
            $step('Aktifkan pengingat pada agenda', (clone $agendas)->whereNotNull('remind_minutes')->exists(), "$base/kalender"),
        ]);

        $out['tiket'] = $make([
            $step('Buat jenis tiket', (clone $types)->exists(), "$base/tiket"),
            $step('Buka penjualan tiket', (bool) $event->ticket_sales_open, "$base/tiket"),
            $step('Dapatkan tiket terjual', (clone $orders)->where('status', 'lunas')->exists(), "$base/tiket"),
        ]);

        $out['voting'] = $make([
            $step('Tambah minimal 2 peserta aktif', $activeParts >= 2, "$base/voting"),
            $step('Lengkapi foto semua peserta', $activeParts > 0 && (clone $parts)->whereNull('photo')->doesntExist(), "$base/voting"),
            $step('Atur periode voting', (bool) ($vs && $vs->vote_start && $vs->vote_end), "$base/voting"),
            $step('Buka voting', (bool) ($vs && $vs->enabled), "$base/voting"),
        ]);

        $reg = [$step('Atur periode pendaftaran', $regSet, "$base/kalender")];
        if ($event->registration_method === 'ticket') {
            $reg[] = $step('Dapatkan pendaftar pertama', (clone $orders)->whereNotIn('status', $closed)->exists(), "$base/registrasi");
            $reg[] = $step('Ada pembayaran terkonfirmasi', (clone $orders)->where('status', 'lunas')->exists(), "$base/registrasi");
        } else {
            $reg[] = $step('Dapatkan pendaftar pertama', $db::table('event_registrations')->where('event_id', $event->id)->exists(), "$base/registrasi");
        }
        $out['registrasi'] = $make($reg);

        $jc = $db::table('judging_criteria')->where('event_id', $event->id);
        $jj = $db::table('judges')->where('event_id', $event->id)->where('active', 1);
        $js = $db::table('judging_settings')->where('event_id', $event->id)->first();
        $out['penilaian'] = $make([
            $step('Atur kriteria tiap kategori (bobot 100%)', \App\Support\Judging::criteriaReady($event->id), "$base/penilaian"),
            $step('Tambah minimal 1 juri aktif', (clone $jj)->exists(), "$base/penilaian"),
            $step('Sahkan semua kategori', \App\Support\Judging::allApproved($event->id), "$base/penilaian"),
            $step('Dapatkan nilai pertama dari juri', $db::table('judging_scores')->where('event_id', $event->id)->exists(), "$base/penilaian"),
        ]);

        return $out;
    }

    /** Pengaturan tiket: jenis, harga, kuota, dan buka/tutup penjualan. */
    private function tiket(\Illuminate\Http\Request $request, Event $event): Response
    {
        $perms = (array) $request->attributes->get('panel_perms', []);
        abort_unless($request->attributes->get('is_owner') || in_array('events', $perms, true), 403);

        $db = \Illuminate\Support\Facades\DB::class;
        $types = \App\Models\TicketType::where('event_id', $event->id)->orderBy('id')->get()->map(function ($t) use ($db) {
            $sold = TicketSettingsController::sold($t->id);

            return [
                'id' => $t->id,
                'name' => $t->name,
                'price' => (int) $t->price,
                'quota' => (int) $t->quota,
                'valid_from' => $t->valid_from->toDateString(),
                'valid_until' => $t->valid_until->toDateString(),
                'valid_days' => (int) $t->valid_days,
                'custom_sale' => (bool) $t->custom_sale,
                'require_holder_details' => (bool) $t->require_holder_details,
                'support_enabled' => (bool) $t->support_enabled,
                'support_options' => json_decode((string) $t->support_options, true) ?: [],
                'support_ranking' => (function () use ($t) {
                    $v = \Illuminate\Support\Facades\DB::table('ticket_orders')
                        ->where('ticket_type_id', $t->id)->where('status', 'lunas')->whereNotNull('support_choice')
                        ->groupBy('support_choice')
                        ->pluck(\Illuminate\Support\Facades\DB::raw('SUM(quantity)'), 'support_choice');

                    return collect(json_decode((string) $t->support_options, true) ?: [])
                        ->map(fn ($n) => ['name' => $n, 'votes' => (int) ($v[$n] ?? 0)])
                        ->sortByDesc('votes')->values()->all();
                })(),
                'max_per_order' => (int) $t->max_per_order,
                'sold' => $sold,
                'remaining' => max(0, (int) $t->quota - $sold),
                'revenue' => (int) $db::table('ticket_orders')->where('ticket_type_id', $t->id)->where('status', 'lunas')->sum('subtotal'),
                'locked' => $db::table('ticket_orders')->where('ticket_type_id', $t->id)->where('status', '!=', 'batal')->exists(),
            ];
        })->values();

        return $this->render('panel/events/tiket', $event, [
            'eventId' => $event->id,
            'salesOpen' => (bool) $event->ticket_sales_open,
            'feePercent' => \App\Models\Setting::serviceFeePercent(),
            'types' => $types,
        ]);
    }

    /** Registrasi peserta untuk event yang pendaftarannya lewat pembelian tiket. */
    private function registrasi(Event $event): Response
    {
        $db = \Illuminate\Support\Facades\DB::class;
        $typeName = $db::table('ticket_types')->where('event_id', $event->id)->pluck('name', 'id');
        $orders = $db::table('ticket_orders')->where('event_id', $event->id)->orderByDesc('id')
            ->get(['id', 'code', 'buyer_name', 'buyer_email', 'ticket_type_id', 'quantity', 'status', 'holders', 'holder_details', 'support_choice']);
        $holders = $db::table('ticket_holders')->where('event_id', $event->id)
            ->get(['ticket_order_id', 'code', 'reg_code', 'holder_name', 'checked_in_at'])->groupBy('ticket_order_id');

        $closed = ['dibatalkan', 'batal', 'kedaluwarsa', 'ditolak'];
        $rows = [];
        $sum = ['orders' => $orders->count(), 'sold' => 0, 'pending' => 0, 'checked_in' => 0];

        foreach ($orders as $o) {
            $type = $typeName[$o->ticket_type_id] ?? '-';
            $list = $holders[$o->id] ?? collect();
            $names = $list->isNotEmpty() ? [] : (json_decode((string) $o->holders, true) ?: []);

            if ($o->status === 'lunas') {
                $sum['sold'] += (int) $o->quantity;
            } elseif (! in_array($o->status, $closed, true)) {
                $sum['pending'] += (int) $o->quantity;
            }

            $base = ['order' => $o->code, 'type' => $type, 'buyer' => $o->buyer_name, 'email' => $o->buyer_email, 'status' => $o->status];
            $det = json_decode((string) $o->holder_details, true) ?: [];
            $contact = fn ($d) => $d ? trim(($d['email'] ?? '').' · '.($d['phone'] ?? ''), ' ·') : null;
            foreach ($list->values() as $i => $h) {
                $in = $h->checked_in_at !== null;
                $sum['checked_in'] += $in ? 1 : 0;
                $rows[] = $base + ['name' => $h->holder_name, 'code' => $h->code, 'reg_id' => $h->reg_code, 'checked_in' => $in, 'contact' => $contact($det[$i] ?? null), 'support' => $o->support_choice];
            }
            foreach ($names as $j => $n) {
                $rows[] = $base + ['name' => (string) $n, 'code' => null, 'reg_id' => null, 'checked_in' => false, 'contact' => $contact($det[$j] ?? null), 'support' => $o->support_choice];
            }
        }

        return $this->render('panel/events/registrasi', $event, [
            'eventId' => $event->id,
            'summary' => $sum,
            'types' => $typeName->values(),
            'rows' => $rows,
        ]);
    }

    private function tiketInduk(Event $event): Response
    {
        $db = \Illuminate\Support\Facades\DB::class;
        $children = $event->children()->get();
        $ids = $children->pluck('id')->all();
        $titles = $children->pluck('title', 'id');
        $today = \Illuminate\Support\Carbon::now(config('app.agenda_timezone', 'Asia/Jakarta'))->toDateString();

        $sold = $db::table('ticket_holders as h')
            ->join('ticket_orders as o', 'o.id', '=', 'h.ticket_order_id')
            ->whereIn('h.event_id', $ids)
            ->where('o.status', 'lunas')
            ->selectRaw('h.ticket_type_id as tid, count(*) as n')
            ->groupBy('h.ticket_type_id')
            ->pluck('n', 'tid');

        $fmt = fn ($d) => $d ? \Illuminate\Support\Carbon::parse($d)->format('d M Y') : null;
        $rows = $db::table('ticket_types')->whereIn('event_id', $ids)->orderBy('event_id')->orderBy('id')->get()
            ->map(function ($t) use ($titles, $sold, $today, $fmt) {
                $n = (int) ($sold[$t->id] ?? 0);
                $quota = (int) $t->quota;
                $from = $t->valid_from ? \Illuminate\Support\Carbon::parse($t->valid_from)->toDateString() : null;
                $until = $t->valid_until ? \Illuminate\Support\Carbon::parse($t->valid_until)->toDateString() : null;
                $status = 'dibuka';
                if ($from && $today < $from) {
                    $status = 'belum';
                } elseif ($until && $today > $until) {
                    $status = 'tutup';
                } elseif ($quota > 0 && $n >= $quota) {
                    $status = 'habis';
                }

                return [
                    'sub_id' => $t->event_id,
                    'sub' => $titles[$t->event_id] ?? '-',
                    'name' => $t->name,
                    'price' => (int) $t->price,
                    'quota' => $quota,
                    'sold' => $n,
                    'revenue' => $n * (int) $t->price,
                    'from' => $fmt($t->valid_from),
                    'until' => $fmt($t->valid_until),
                    'status' => $status,
                ];
            })->values();

        return $this->render('panel/events/tiket-induk', $event, [
            'subs' => $children->map(fn ($c) => [
                'id' => $c->id,
                'title' => $c->title,
                'method' => $c->registration_method,
            ])->values(),
            'rows' => $rows,
        ]);
    }

    private function laporan(Event $event): Response
    {
        $db = \Illuminate\Support\Facades\DB::class;
        $isParent = $event->parent_id === null && $event->children()->exists();
        $events = $isParent ? $event->children()->get() : collect([$event]);
        $ids = $events->pluck('id')->all();
        $titles = $events->pluck('title', 'id');
        $tz = config('app.agenda_timezone', 'Asia/Jakarta');
        $closed = ['dibatalkan', 'batal', 'kedaluwarsa', 'ditolak'];
        $grp = fn ($s) => $s === 'lunas' ? 'lunas' : (in_array($s, $closed, true) ? 'batal' : 'menunggu');
        $day = fn ($d) => \Illuminate\Support\Carbon::parse($d)->setTimezone($tz)->format('Y-m-d');

        $perSub = [];
        foreach ($ids as $id) {
            $perSub[$id] = ['id' => $id, 'title' => $titles[$id] ?? '-', 'total' => 0, 'lunas' => 0, 'menunggu' => 0, 'batal' => 0, 'revenue' => 0, 'tickets' => 0, 'hadir' => 0];
        }
        $typeRev = [];
        $trend = [];
        $catCount = [];

        $types = $db::table('ticket_types')->whereIn('event_id', $ids)->orderBy('event_id')->orderBy('id')->get(['id', 'event_id', 'name', 'price', 'quota']);
        $cats = $db::table('event_participant_categories')->whereIn('event_id', $ids)->pluck('name', 'id');

        $orders = $db::table('ticket_orders')->whereIn('event_id', $ids)
            ->get(['event_id', 'ticket_type_id', 'status', 'quantity', 'subtotal', 'created_at', 'paid_at']);
        foreach ($orders as $o) {
            $g = $grp((string) $o->status);
            $q = max(1, (int) $o->quantity);
            $perSub[$o->event_id][$g] += $q;
            if ($g === 'lunas') {
                $perSub[$o->event_id]['revenue'] += (int) $o->subtotal;
                $typeRev[$o->ticket_type_id] = ($typeRev[$o->ticket_type_id] ?? 0) + (int) $o->subtotal;
                $k = $day($o->paid_at ?: $o->created_at);
                $trend[$k] = ($trend[$k] ?? 0) + $q;
            }
        }

        $regs = $db::table('event_registrations')->whereIn('event_id', $ids)
            ->get(['event_id', 'status', 'total_fee', 'participant_category_id', 'created_at']);
        foreach ($regs as $r) {
            $g = $grp((string) $r->status);
            $perSub[$r->event_id][$g] += 1;
            if ($g === 'lunas') {
                $perSub[$r->event_id]['revenue'] += (int) $r->total_fee;
                $k = $day($r->created_at);
                $trend[$k] = ($trend[$k] ?? 0) + 1;
            }
            if ($g !== 'batal') {
                $n = $cats[$r->participant_category_id] ?? 'Tanpa kategori';
                $catCount[$n] = ($catCount[$n] ?? 0) + 1;
            }
        }

        $sold = [];
        $hadirType = [];
        $holders = $db::table('ticket_holders as h')
            ->join('ticket_orders as o', 'o.id', '=', 'h.ticket_order_id')
            ->whereIn('h.event_id', $ids)->where('o.status', 'lunas')
            ->selectRaw('h.event_id as eid, h.ticket_type_id as tid, count(*) as n, count(h.checked_in_at) as c')
            ->groupBy('h.event_id', 'h.ticket_type_id')->get();
        foreach ($holders as $h) {
            $sold[$h->tid] = ($sold[$h->tid] ?? 0) + (int) $h->n;
            $hadirType[$h->tid] = ($hadirType[$h->tid] ?? 0) + (int) $h->c;
            $perSub[$h->eid]['tickets'] += (int) $h->n;
            $perSub[$h->eid]['hadir'] += (int) $h->c;
        }

        foreach ($perSub as &$p) {
            $p['total'] = $p['lunas'] + $p['menunggu'];
        }
        unset($p);
        $sum = ['total' => 0, 'lunas' => 0, 'menunggu' => 0, 'batal' => 0, 'revenue' => 0, 'tickets' => 0, 'hadir' => 0];
        foreach ($perSub as $p) {
            foreach ($sum as $k => $_) {
                $sum[$k] += $p[$k];
            }
        }

        ksort($trend);
        $trendOut = [];
        foreach (array_slice($trend, -30, null, true) as $d => $n) {
            $trendOut[] = ['date' => \Illuminate\Support\Carbon::parse($d)->format('d M'), 'n' => $n];
        }

        return $this->render('panel/events/laporan', $event, [
            'isParent' => $isParent,
            'summary' => $sum,
            'subs' => array_values($perSub),
            'types' => $types->map(fn ($t) => [
                'sub' => $titles[$t->event_id] ?? '-',
                'name' => $t->name,
                'price' => (int) $t->price,
                'quota' => (int) $t->quota,
                'sold' => (int) ($sold[$t->id] ?? 0),
                'hadir' => (int) ($hadirType[$t->id] ?? 0),
                'revenue' => (int) ($typeRev[$t->id] ?? 0),
            ])->values(),
            'cats' => collect($catCount)->map(fn ($n, $k) => ['name' => $k, 'n' => $n])->values(),
            'trend' => $trendOut,
        ]);
    }

    /** Registrasi event utama: gabungan semua sub-event, tiap sub-event mengikuti metodenya (tiket atau formulir transfer manual). */
    private function registrasiInduk(Event $event): Response
    {
        $db = \Illuminate\Support\Facades\DB::class;
        $tz = config('app.agenda_timezone', 'Asia/Jakarta');
        $children = $event->children()->get();
        $ids = $children->pluck('id')->all();
        $titles = $children->pluck('title', 'id');
        $typeName = $db::table('ticket_types')->whereIn('event_id', $ids)->pluck('name', 'id');
        $cats = $db::table('event_participant_categories')->whereIn('event_id', $ids)->pluck('name', 'id');
        $norm = function ($s) {
            $s = (string) $s;
            if (in_array($s, ['lunas', 'terdaftar'], true)) {
                return 'terdaftar';
            }
            if ($s === 'ditolak') {
                return 'ditolak';
            }
            if (in_array($s, ['dibatalkan', 'batal', 'kedaluwarsa'], true)) {
                return 'batal';
            }

            return 'menunggu';
        };
        $fmt = fn ($d) => $d ? \Illuminate\Support\Carbon::parse($d)->setTimezone($tz)->format('d M Y H:i') : '';
        $contact = fn ($d) => $d ? trim(($d['email'] ?? '').' · '.($d['phone'] ?? ''), ' ·') : null;
        $rows = [];
        $sum = [];
        foreach ($children as $c) {
            $sum[$c->id] = ['orders' => 0, 'sold' => 0, 'pending' => 0];
        }

        // Sub-event bertiket: pemegang tiket (hanya dilihat).
        $orders = $db::table('ticket_orders')->whereIn('event_id', $ids)->orderByDesc('id')
            ->get(['id', 'event_id', 'code', 'ticket_type_id', 'buyer_name', 'buyer_email', 'status', 'quantity', 'holders', 'holder_details', 'support_choice', 'created_at']);
        $holders = $db::table('ticket_holders')->whereIn('event_id', $ids)
            ->get(['ticket_order_id', 'code', 'reg_code', 'holder_name'])->groupBy('ticket_order_id');
        foreach ($orders as $o) {
            $st = $norm($o->status);
            $sum[$o->event_id]['orders']++;
            if ($st === 'terdaftar') {
                $sum[$o->event_id]['sold'] += (int) $o->quantity;
            } elseif ($st === 'menunggu') {
                $sum[$o->event_id]['pending'] += (int) $o->quantity;
            }
            $list = ($holders[$o->id] ?? collect())->values();
            $det = json_decode((string) $o->holder_details, true) ?: [];
            $base = [
                'kind' => 'tiket',
                'id' => null,
                'sub_id' => $o->event_id,
                'sub' => $titles[$o->event_id] ?? '-',
                'order' => $o->code,
                'buyer' => $o->buyer_name,
                'type' => $typeName[$o->ticket_type_id] ?? '-',
                'support' => $o->support_choice,
                'email' => (string) $o->buyer_email,
                'phone' => null,
                'school' => null,
                'category' => $typeName[$o->ticket_type_id] ?? null,
                'status' => $st,
                'amount' => 0,
                'proof' => false,
                'note' => null,
                'at' => $fmt($o->created_at),
                'followup_done' => false,
                'teams' => [],
                'answers' => (object) [],
            ];
            foreach ($list as $i => $h) {
                $rows[] = $base + ['reg_id' => $h->reg_code, 'name' => $h->holder_name ?: '-', 'code' => $h->code, 'contact' => $contact($det[$i] ?? null)];
            }
            if ($list->isEmpty()) {
                foreach ((json_decode((string) $o->holders, true) ?: []) as $j => $n) {
                    $rows[] = $base + ['reg_id' => null, 'name' => (string) $n, 'code' => null, 'contact' => $contact($det[$j] ?? null)];
                }
            }

            return $this->tiketInduk($event);

        if ($menu === 'laporan') {
            return $this->laporan($event);
        }

        if ($menu === 'tiket' && $event->registration_method === 'ticket') {
            return $this->tiket($request, $event);
        }

        if ($menu === 'checkin' && $event->registration_method === 'ticket') {
            return $this->checkin($event);
        }

        if ($menu === 'registrasi' && $event->registration_method === 'ticket') {
            return $this->registrasi($event);

            return $this->registrasiForm($event);
        }

        return $this->render('panel/events/menu', $event, [
            'menu' => ['key' => $menu, 'label' => $label],
        ]);
    }

class TicketSupportController extends Controller
{
    private function guard(Request $request, Event $event): void
    {
        $inst = $request->attributes->get('institution');
        abort_unless($inst && $event->institution_id === $inst->id, 404);
        abort_unless($event->registration_method === 'ticket' && $event->status === Event::PUBLISHED, 404);
    }

    /** Daftar pilihan + jumlah dukungan (tiket lunas), urut terbanyak. */
    public static function ranking(int $eventId): array
    {
        return DB::table('ticket_support_options as o')
            ->leftJoin('ticket_orders as t', fn ($j) => $j->on('t.support_option_id', '=', 'o.id')->where('t.status', 'lunas'))
            ->where('o.event_id', $eventId)
            ->groupBy('o.id', 'o.name')
            ->orderByRaw('COALESCE(SUM(t.quantity),0) desc')->orderBy('o.name')
            ->get(['o.id', 'o.name', DB::raw('COALESCE(SUM(t.quantity),0) as votes')])
            ->map(fn ($r) => ['id' => (int) $r->id, 'name' => $r->name, 'votes' => (int) $r->votes])
            ->all();
    }

    public function index(Request $request, Event $event): JsonResponse
    {
        $this->guard($request, $event);

        return response()->json(['options' => self::ranking($event->id)]);
    }

    public function store(Request $request, Event $event): RedirectResponse
    {
        $this->guard($request, $event);
        $data = $request->validate(['name' => ['required', 'string', 'max:80']], [
            'name.required' => 'Isi nama pilihan dukungan.',
        ]);
        $name = trim($data['name']);

        if (DB::table('ticket_support_options')->where('event_id', $event->id)->whereRaw('lower(name) = ?', [mb_strtolower($name)])->exists()) {
            return back()->withErrors(['name' => 'Pilihan dukungan itu sudah ada.']);
        }
        if (DB::table('ticket_support_options')->where('event_id', $event->id)->count() >= 50) {
            return back()->withErrors(['name' => 'Maksimal 50 pilihan dukungan.']);
        }

        DB::table('ticket_support_options')->insert(['event_id' => $event->id, 'name' => $name, 'created_at' => now(), 'updated_at' => now()]);

        return back()->with('success', 'Pilihan dukungan ditambahkan.');
    }

    public function destroy(Request $request, Event $event, int $option): RedirectResponse
    {
        $this->guard($request, $event);
        abort_unless(DB::table('ticket_support_options')->where('id', $option)->where('event_id', $event->id)->exists(), 404);

        if (DB::table('ticket_orders')->where('support_option_id', $option)->where('status', '!=', 'batal')->exists()) {
            return back()->with('error', 'Pilihan yang sudah didukung pembeli tidak bisa dihapus.');
        }

        DB::table('ticket_support_options')->where('id', $option)->delete();

        return back()->with('success', 'Pilihan dukungan dihapus.');
    }

    public function update(Request $request, Event $event): RedirectResponse
    {
        $this->own($request, $event);

        $d = $request->validate([
            'fields' => ['present', 'array', 'max:200'],
            'fields.*.key' => ['nullable', 'string', 'max:20'],
            'fields.*.label' => ['required', 'string', 'max:120'],
            'fields.*.type' => ['required', Rule::in(RegistrationForm::TYPES)],
            'fields.*.required' => ['boolean'],
            'fields.*.stage' => ['nullable', Rule::in(['register', 'after'])],
            'fields.*.options' => ['nullable', 'array', 'max:50'],
            'fields.*.options.*' => ['string', 'max:80'],
            'message' => ['nullable', 'string', 'max:2000'],
        ], [], ['fields.*.label' => 'Label kolom']);

        $existing = collect(\App\Support\FormFields::all($event))->keyBy('key');
        $taken = $existing->keys()->all();
        $out = [];
        $seen = [];

        foreach ($d['fields'] as $i => $f) {
            $type = $f['type'];
            $options = [];
            if (RegistrationForm::needsOptions($type)) {
                $options = array_values(array_unique(array_filter(
                    array_map('trim', $f['options'] ?? []),
                    fn ($o) => $o !== ''
                )));
                if (! $options) {
                    throw ValidationException::withMessages([
                        "fields.$i.options" => 'Kolom "'.trim($f['label']).'" butuh minimal satu pilihan.',
                    ]);
                }
            }

            $key = $f['key'] ?? null;
            if (! $key || ! $existing->has($key) || isset($seen[$key])) {
                do {
                    $key = 'f_'.Str::lower(Str::random(8));
                } while (in_array($key, $taken, true) || isset($seen[$key]));
            }
            $seen[$key] = true;

            $out[] = [
                'key' => $key,
                'label' => trim($f['label']),
                'type' => $type,
                'required' => (bool) ($f['required'] ?? false),
                'stage' => (($f['stage'] ?? 'register') === 'after') ? 'after' : 'register',
                'options' => $options,
                'archived' => false,
            ];
        }

        // Kolom yang dihapus tapi sudah punya jawaban diarsipkan, supaya jawaban lama tidak hilang.
        if (DB::table('event_registrations')->where('event_id', $event->id)->where(fn ($q) => $q->whereNotNull('answers')->orWhereNotNull('followup_answers'))->exists()) {
            foreach ($existing as $key => $f) {
                if (! isset($seen[$key])) {
                    $out[] = ['archived' => true] + $f;
                }
            }
        }

        DB::table('events')->where('id', $event->id)->update([
            'registration_fields' => json_encode($out, JSON_UNESCAPED_UNICODE),
            'registration_message' => $d['message'] ?? null,
        ]);

        return back();
    }

    public function review(Request $request, Event $event, int $reg): RedirectResponse
    {
        $this->own($request, $event);

        $d = $request->validate([
            'status' => ['required', Rule::in(['menunggu', 'terdaftar', 'ditolak'])],
            'note' => ['nullable', 'string', 'max:500'],
        ]);
        if ($d['status'] === 'ditolak' && blank($d['note'] ?? null)) {
            throw ValidationException::withMessages(['note' => 'Alasan penolakan wajib diisi.']);
        }

        $n = DB::table('event_registrations')->where('event_id', $event->id)->where('id', $reg)->update([
            'status' => $d['status'],
            'review_note' => $d['status'] === 'ditolak' ? trim($d['note']) : null,
            'reviewed_at' => now(),
            'updated_at' => now(),
        ]);
        abort_unless($n, 404);

        if ($d['status'] !== 'menunggu') {
            try {
                $row = DB::table('event_registrations')->where('id', $reg)->first();
                $to = $row ? \App\Models\User::find($row->user_id) : null;

            ->when(isset($cats[$type]), fn ($q) => $q->where(fn ($w) => $w->where('category', $type)
                ->orWhereHas('children', fn ($c) => $c->where('category', $type))))
            ->orderBy('starts_at')->orderBy('id')
            ->paginate(10)->withQueryString()
            ->through(fn (Event $e) => [
                'id' => $e->id,
                'title' => $e->title,
                'category' => $e->category,
                'category_label' => $cats[$e->category] ?? ($e->category ?: '-'),
                'location' => $e->location,
                'starts_at' => $e->starts_at->toDateString(),
                'ends_at' => $e->ends_at->toDateString(),
                'sub_count' => $e->children_count,
                'institution' => $e->institution?->name,
            ]);

        return Inertia::render('admin/manajemen-event', [
            'events' => $events,
            'filters' => ['search' => $search, 'type' => $type],
            'counts' => collect($cats)->keys()
                ->mapWithKeys(fn ($k) => [$k => Event::where('status', Event::PUBLISHED)
                    ->where('category', $k)
                    ->where(fn ($w) => $w->whereNull('parent_id')
                        ->orWhereIn('parent_id', fn ($s) => $s->select('id')->from('events')->where('status', Event::PUBLISHED)))
                    ->count()])
                ->prepend($base()->count(), 'total')
                ->all(),
            'categories' => $cats,
        ]);
    }

    public function update(Request $request, Event $event): RedirectResponse
    {
        abort_if($event->parent_id !== null, 404);

        $data = $request->validate([
            'title' => ['required', 'string', 'max:150'],
            'category' => ['nullable', 'string', 'in:'.implode(',', array_keys(PanelEventController::CATEGORIES))],
            'location' => ['nullable', 'string', 'max:200'],
            'starts_at' => ['required', 'date'],
            'ends_at' => ['required', 'date', 'after_or_equal:starts_at'],
        ]);

        // Rangkaian tidak punya kategori sendiri
        if ($event->children()->exists()) {
            $data['category'] = $event->category;
        }

        $event->forceFill($data)->save();

        return back()->with('success', 'Event berhasil diperbarui.');
    }

    /** Validasi dan simpan edit penuh. $claim dijalankan di dalam transaksi sebelum menyimpan. */
    public function perform(Request $request, Event $event, ?\Closure $claim = null): string
    {
        abort_if($event->parent_id !== null, 404);

        $children = $event->children()->get()->keyBy('id');
        $series = $children->isNotEmpty();
        $panel = app(\App\Http\Controllers\Panel\EventController::class);
        $maxCats = (int) ($event->package?->max_participant_categories ?: 100);

        // Kolom yang dikunci selalu diambil dari data tersimpan, bukan dari kiriman form.
        $in = $request->input();
        if ($series) {
            foreach ((array) ($in['subs'] ?? []) as $i => $s) {
                $c = $children->get((int) ($s['id'] ?? 0));
                abort_if($c === null, 422);
                $in['subs'][$i] = array_merge($s, $this->locked($c));
            }
        } else {
            $in = array_merge($in, $this->locked($event));
        }
        $request->merge($in);

        $base = fn (string $p) => [
            "{$p}title" => ['required', 'string', 'max:150'],
            "{$p}description" => ['nullable', 'string', 'max:2000'],
            "{$p}starts_at" => ['required', 'date'],
            "{$p}ends_at" => ['required', 'date', "after_or_equal:{$p}starts_at"],
            "{$p}poster" => ['nullable', 'image', 'mimes:jpg,jpeg,png,webp', 'max:30720'],
            "{$p}juknis" => ['nullable', 'file', 'mimes:pdf', 'max:5120'],
            "{$p}participant_categories.*.id" => ['nullable', 'integer'],
            "{$p}ticket_types.*.id" => ['nullable', 'integer'],
        ];

        if ($series) {
            $rules = $base('') + ['subs' => ['required', 'array'], 'subs.*.id' => ['required', 'integer']]
                + $base('subs.*.') + $panel->detailRules('subs.*.', $maxCats);
        } else {
            $rules = $base('') + $panel->detailRules('', $maxCats);
        }

        $names = [
            'title' => 'Nama event', 'category' => 'Kategori event', 'location' => 'Lokasi',
            'starts_at' => 'Tanggal mulai', 'ends_at' => 'Tanggal selesai', 'description' => 'Deskripsi',
            'poster' => 'Poster', 'juknis' => 'Juknis', 'participant_categories' => 'Kategori peserta',
            'participant_categories.*.name' => 'Nama kategori peserta', 'participant_categories.*.quota' => 'Kuota kategori',
            'participant_categories.*.competition_name' => 'Nama perlombaan/pertandingan',
            'participant_categories.*.min_age' => 'Usia minimal', 'participant_categories.*.max_age' => 'Usia maksimal',
            'participant_categories.*.requirements' => 'Syarat tambahan',
            'participant_categories.*.prizes.*.rank' => 'Juara', 'participant_categories.*.prizes.*.prize' => 'Hadiah',
            'registration_fee' => 'Biaya pendaftaran', 'ticket_types' => 'Jenis tiket', 'ticket_types.*.name' => 'Nama tiket',
            'ticket_types.*.price' => 'Harga tiket', 'ticket_types.*.quota' => 'Kuota tiket',
            'ticket_types.*.valid_from' => 'Tanggal mulai berlaku', 'ticket_types.*.valid_until' => 'Tanggal akhir berlaku',
            'ticket_types.*.max_per_order' => 'Maksimal tiket per pesanan', 'competition_type' => 'Jenis perlombaan',
            'team_size' => 'Jumlah peserta per tim', 'team_per_row' => 'Peserta per baris', 'team_leaders' => 'Kotak ketua',
            'bank_name' => 'Nama bank', 'bank_account_number' => 'Nomor rekening', 'bank_account_name' => 'Nama pemilik rekening',
            'contact_name' => 'Nama penanggung jawab', 'contact_phone' => 'Nomor kontak', 'contact_email' => 'Email', 'contact_position' => 'Jabatan',
        ];
        $attributes = [];
        foreach ($names as $k => $v) {
            $attributes[$k] = $v;
            $attributes["subs.*.$k"] = $v;
        }
        $messages = [
            'required' => ':attribute wajib diisi.',
            'integer' => ':attribute harus berupa angka.',
            'email' => 'Format :attribute tidak valid.',
            'image' => ':attribute harus berupa gambar.',
            'date' => ':attribute tidak valid.',
            'in' => ':attribute tidak valid.',
            'min.numeric' => ':attribute minimal :min.',
            'min.array' => 'Tambahkan minimal :min :attribute.',
            'max.array' => ':attribute maksimal :max.',
            'max.string' => ':attribute terlalu panjang (maksimal :max karakter).',
            'max.file' => 'Ukuran :attribute terlalu besar.',
            'ends_at.after_or_equal' => 'Tanggal selesai tidak boleh sebelum tanggal mulai.',
            'subs.*.ends_at.after_or_equal' => 'Tanggal selesai tidak boleh sebelum tanggal mulai.',
        ];

        $data = $request->validate($rules, $messages, $attributes);

        $comp = ['perlombaan', 'olahraga'];
        $check = function (array $d, string $pre) use ($comp) {
            $n = array_map(fn ($c) => mb_strtolower(trim($c['name'])), $d['participant_categories']);
            if (count($n) !== count(array_unique($n))) {
                $this->bad($pre.'participant_categories', 'Ada kategori peserta yang sama.');
            }
            if (in_array($d['category'] ?? '', $comp, true)) {
                if (empty($d['competition_type'])) {
                    $this->bad($pre.'competition_type', 'Pilih perlombaan perorangan atau tim.');
                }
                if (($d['competition_type'] ?? null) === 'team') {
                    if ((int) ($d['team_size'] ?? 0) < 2) {
                        $this->bad($pre.'team_size', 'Jumlah anggota per tim minimal 2.');
                    }
                    if ((int) ($d['team_leaders'] ?? 0) >= (int) ($d['team_size'] ?? 0)) {
                        $this->bad($pre.'team_leaders', 'Jumlah kotak ketua harus lebih kecil dari jumlah peserta per tim.');
                    }
                }
            }
        };
        if ($series) {
            foreach ($data['subs'] as $i => $s) {
                $check($s, "subs.$i.");
                if ($s['starts_at'] < $data['starts_at']) {
                    $this->bad("subs.$i.starts_at", 'Tanggal tidak boleh sebelum event utama dimulai.');
                }
                if ($s['ends_at'] > $data['ends_at']) {
                    $this->bad("subs.$i.ends_at", 'Tanggal tidak boleh melewati event utama selesai.');
                }
            }
        } else {
            $check($data, '');
        }

        $trash = \Illuminate\Support\Facades\DB::transaction(function () use ($event, $children, $series, $data, $request, $claim) {
            if ($claim) {
                $claim();
            }
            $old = $this->saveEvent($event, $data, $request, '', ! $series);
            if ($series) {
                foreach ($data['subs'] as $i => $s) {
                    $old = array_merge($old, $this->saveEvent($children[(int) $s['id']], $s, $request, "subs.$i.", true));
                }
            }

            return $old;
        });

        foreach ($trash as $path) {
            \Illuminate\Support\Facades\Storage::disk('public')->delete($path);
        }

        return $data['title'];
    }

    {
        throw \Illuminate\Validation\ValidationException::withMessages([$key => $msg]);
    }

    {

            ];
        }

        return $f;
    }

    /** @return array<int, string> path file lama yang boleh dihapus setelah commit */
    private function saveEvent(Event $e, array $d, Request $request, string $pre, bool $detail): array
    {
        $trash = [];
        $attrs = [
            'title' => $d['title'],
            'description' => $d['description'] ?? null,
            'starts_at' => $d['starts_at'],
            'ends_at' => $d['ends_at'],
        ];
        if ($request->hasFile($pre.'poster')) {
            $attrs['poster_path'] = \App\Support\ImageUploader::store($request->file($pre.'poster'), 'posters', $pre.'poster');
            if ($e->poster_path) {
                $trash[] = $e->poster_path;
            }
        }
        if ($request->hasFile($pre.'juknis')) {
            $attrs['juknis_path'] = $request->file($pre.'juknis')->store('juknis', 'public');
            if ($e->juknis_path) {
                $trash[] = $e->juknis_path;
            }
        }

        $isComp = $detail && in_array($d['category'] ?? '', ['perlombaan', 'olahraga'], true);
        if ($detail) {
            $team = $isComp && ($d['competition_type'] ?? null) === 'team';
            $manual = $e->registration_method === 'transfer_manual';
            $attrs += [
                'category' => $d['category'],
                'location' => $d['location'],
                'participant_quota' => (int) collect($d['participant_categories'])->sum(fn ($c) => (int) $c['quota']),
                'competition_type' => $isComp ? ($d['competition_type'] ?? null) : null,
                'team_size' => $team ? (int) $d['team_size'] : null,
                'team_per_row' => $team ? max(1, (int) ($d['team_per_row'] ?? 1)) : null,
                'team_leaders' => $team ? (int) ($d['team_leaders'] ?? 0) : null,
                'team_slots' => $team
                    ? json_encode(collect(range(1, (int) $d['team_size']))->map(fn ($n) => trim((string) ($d['team_slots'][$n - 1] ?? '')) ?: 'Peserta '.$n)->all(), JSON_UNESCAPED_UNICODE)
                    : null,
                'registration_fee' => (int) ($d['registration_fee'] ?? 0),
                'bank_name' => $manual ? ($d['bank_name'] ?? null) : null,
                'bank_account_number' => $manual ? ($d['bank_account_number'] ?? null) : null,
                'bank_account_name' => $manual ? ($d['bank_account_name'] ?? null) : null,
                'contact_name' => $d['contact_name'],
                'contact_phone' => $d['contact_phone'],
                'contact_email' => $d['contact_email'] ?? null,
                'contact_position' => $d['contact_position'] ?? null,
            ];
        }
        $e->forceFill($attrs)->save();

        if ($detail) {
            $this->syncCategories($e, $d['participant_categories'], $isComp, $pre);
            if ($e->registration_method === 'ticket') {
                $this->syncTickets($e, (array) ($d['ticket_types'] ?? []), $pre);
            }
        }

        return $trash;
    }

    private function syncCategories(Event $e, array $cats, bool $comp, string $pre): void
    {
        $existing = $this->tbl('event_participant_categories')->where('event_id', $e->id)->get()->keyBy('id');
        $keep = [];
        $now = now();

        foreach (array_values($cats) as $i => $c) {
            $prizes = array_values(array_filter(array_map(fn ($p) => [
                'rank' => trim((string) ($p['rank'] ?? '')),
                'prize' => trim((string) ($p['prize'] ?? '')),
            ], $c['prizes'] ?? []), fn ($p) => $p['rank'] !== ''));

            $row = [
                'name' => trim($c['name']),
                'quota' => (int) $c['quota'],
                'competition_name' => $comp ? (trim((string) ($c['competition_name'] ?? '')) ?: null) : null,
                'min_age' => $comp && ($c['min_age'] ?? '') !== '' ? (int) $c['min_age'] : null,
                'max_age' => $comp && ($c['max_age'] ?? '') !== '' ? (int) $c['max_age'] : null,
                'requirements' => $comp ? (trim((string) ($c['requirements'] ?? '')) ?: null) : null,
                'prizes' => $comp && $prizes ? json_encode($prizes) : null,
                'updated_at' => $now,
            ];

            $id = (int) ($c['id'] ?? 0);
            if ($id && $existing->has($id)) {
                $reg = $this->tbl('event_registrations')->where('participant_category_id', $id)->count();
                if ((int) $c['quota'] < $reg) {
                    $this->bad("{$pre}participant_categories.$i.quota", "Kuota tidak boleh di bawah jumlah pendaftar ($reg).");
                }
                $this->tbl('event_participant_categories')->where('id', $id)->update($row);
                $keep[] = $id;
            } else {
                $this->tbl('event_participant_categories')->insert($row + ['event_id' => $e->id, 'created_at' => $now]);
            }
        }

        foreach ($existing as $id => $old) {
            if (in_array((int) $id, $keep, true)) {
                continue;
            }
            if ($this->tbl('event_registrations')->where('participant_category_id', $id)->exists()) {
                $this->bad("{$pre}participant_categories", "Kategori \"{$old->name}\" sudah punya pendaftar dan tidak bisa dihapus.");
            }
            $this->tbl('event_participant_categories')->where('id', $id)->delete();
        }
    }

    private function syncTickets(Event $e, array $tix, string $pre): void
    {
        $existing = $this->tbl('ticket_types')->where('event_id', $e->id)->get()->keyBy('id');
        $keep = [];
        $now = now();

        foreach (array_values($tix) as $i => $t) {
            $row = [
                'name' => trim($t['name']),
                'price' => (int) $t['price'],
                'quota' => (int) $t['quota'],
                'valid_from' => $t['valid_from'],
                'valid_until' => $t['valid_until'],
                'max_per_order' => (int) ($t['max_per_order'] ?? 10),
                'updated_at' => $now,
            ];

            $id = (int) ($t['id'] ?? 0);
            if ($id && $existing->has($id)) {
                $orders = $this->tbl('ticket_orders')->where('ticket_type_id', $id)->count();
                $sold = (int) $this->tbl('ticket_orders')->where('ticket_type_id', $id)
                    ->whereIn('status', ['lunas', 'menunggu', 'menunggu_verifikasi'])->sum('quantity');
                if ($orders > 0 && (int) $t['price'] !== (int) $existing[$id]->price) {
                    $this->bad("{$pre}ticket_types.$i.price", 'Harga tidak bisa diubah karena tiket ini sudah punya pesanan.');
                }
                if ((int) $t['quota'] < $sold) {
                    $this->bad("{$pre}ticket_types.$i.quota", "Kuota tidak boleh di bawah jumlah tiket terpesan ($sold).");
                }
                $this->tbl('ticket_types')->where('id', $id)->update($row);
                $keep[] = $id;
            } else {
                $this->tbl('ticket_types')->insert($row + ['event_id' => $e->id, 'created_at' => $now]);
            }
        }

        foreach ($existing as $id => $old) {
            if (in_array((int) $id, $keep, true)) {
                continue;
            }
            if ($this->tbl('ticket_orders')->where('ticket_type_id', $id)->exists()) {
                $this->bad("{$pre}ticket_types", "Tiket \"{$old->name}\" sudah punya pesanan dan tidak bisa dihapus.");
            }
            $this->tbl('ticket_types')->where('id', $id)->delete();
        }
    }

    public function destroy(Event $event): RedirectResponse
    {
        abort_if($event->parent_id !== null, 404);

        $ids = Event::where('parent_id', $event->id)->pluck('id')->push($event->id)->all();

        $hasPaidTickets = \Illuminate\Support\Facades\DB::table('ticket_orders')->whereIn('event_id', $ids)->where('status', 'lunas')->exists();
        $hasWithdrawals = \Illuminate\Support\Facades\DB::table('withdrawals')->whereIn('event_id', $ids)->exists();
        $packagePaid = $event->payment_status === 'lunas';

        if ($hasPaidTickets || $hasWithdrawals || $packagePaid) {
            return back()->with('error', 'Event tidak bisa dihapus karena sudah punya pesanan tiket lunas, pembayaran paket lunas, atau permintaan penarikan. Menghapusnya akan merusak data pendapatan dan transaksi.');
        }

        \Illuminate\Support\Facades\DB::transaction(function () use ($event) {
            Event::where('parent_id', $event->id)->delete();
            $event->delete();
        });

        return back()->with('success', 'Event berhasil dihapus.');
    }

    /** Review Event: antrean event menunggu yang tidak perlu verifikasi pembayaran (gratis, bagi hasil, Midtrans). */
    public function review(Request $request): Response
    {
        return $this->list($request, 'pending', '/admin/review', 'Review Event', true);
    }

    public function approve(Event $event): RedirectResponse
    {
        if ($event->payment_status === 'menunggu_verifikasi') {
            return redirect()->route('admin.verifikasi-paket')
                ->with('success', 'Event ini menunggu verifikasi pembayaran. Setujui di sini, event akan otomatis tayang.');
        }

        // Event utama dan semua kegiatan di dalamnya ditayangkan sekaligus
        Event::where(fn ($q) => $q->where('id', $event->id)->orWhere('parent_id', $event->id))->update([
            'status' => Event::PUBLISHED,
            'rejection_reason' => null,
            'reviewed_at' => now(),
        ]);

        return back()->with('success', 'Event disetujui dan sudah tayang.');
    }

    public function reject(Request $request, Event $event): RedirectResponse
    {
        if ($event->payment_status === 'menunggu_verifikasi') {
            return redirect()->route('admin.verifikasi-paket')
                ->with('success', 'Event ini menunggu verifikasi pembayaran. Setujui di sini, event akan otomatis tayang.');
        }

        $data = $request->validate(['reason' => ['required', 'string', 'max:500']], [
            'reason.required' => 'Alasan penolakan wajib diisi.',
        ]);

        Event::where(fn ($q) => $q->where('id', $event->id)->orWhere('parent_id', $event->id))->update([
            'status' => Event::REJECTED,
            'rejection_reason' => $data['reason'],
            'reviewed_at' => now(),

                        'event_registration_id' => $id,
                        'name' => $t['name'],
                        'participant_category_id' => $t['cat'],
                        'created_at' => now(),
                        'updated_at' => now(),
                    ]);
                }
            });
        } catch (\Throwable $e) {
            Storage::disk('local')->delete($stored);
            throw $e instanceof UniqueConstraintViolationException
                ? ValidationException::withMessages(['form' => 'Anda sudah mendaftar di event ini.'])
                : $e;
        }

        $url = '/events/'.$event->id.'/daftar';
        $this->alert(
            $user,
            $status === 'terdaftar' ? 'Pendaftaran berhasil' : 'Pendaftaran terkirim',
            $status === 'terdaftar'
                ? 'Anda sudah terdaftar di "'.$event->title.'".'
                : 'Pendaftaran Anda di "'.$event->title.'" menunggu verifikasi panitia.',
            $url
        );
        $this->alert(
            \App\Models\Institution::find($event->institution_id)?->user,
            'Pendaftar baru',
            trim($v['contact_name']).' mendaftar di "'.$event->title.'".',
            '/panel/event/'.$event->id.'/registrasi'
        );

        return redirect($url.'?terkirim=1');
    }

    /** Tahap setelah disetujui: pendaftar melengkapi kolom yang diminta panitia. */
    public function complete(Request $request, Event $event): RedirectResponse
    {
        $this->guard($event);
        abort_if($event->competition_type === 'team', 404);
        $user = $request->user();

        $mine = DB::table('event_registrations')->where('event_id', $event->id)->where('user_id', $user->id)->first();
        abort_unless($mine && $mine->status === 'terdaftar', 404);

        $fields = FormFields::active($event, 'after');
        abort_if(! $fields, 404);

        $old = json_decode((string) $mine->followup_answers, true) ?: [];
        [$rules, $msgs, $names] = FormFields::rules($fields, $old);
        $v = $request->validate($rules, $msgs, $names);

        $answers = $old;
        $stored = [];
        $replaced = [];

        try {
            DB::transaction(function () use ($event, $mine, $fields, $v, &$answers, &$stored, &$replaced) {
                foreach ($fields as $f) {
                    $key = $f['key'];
                    $val = data_get($v, 'answers.'.$key);
                    if ($f['type'] === 'file') {
                        if ($val instanceof UploadedFile) {
                            if (isset($answers[$key]['path'])) {
                                $replaced[] = $answers[$key]['path'];
                            }
                            $answers[$key] = $this->keep($val, $event, $stored);
                        }

                        continue;
                    }
                    if ($val === null || $val === '' || $val === []) {
                        unset($answers[$key]);

                        continue;
                    }
                    $answers[$key] = $f['type'] === 'number' ? $val + 0 : (is_array($val) ? array_values($val) : $val);
                }

                DB::table('event_registrations')->where('id', $mine->id)->update([
                    'followup_answers' => $answers ? json_encode($answers, JSON_UNESCAPED_UNICODE) : null,
                    'completed_at' => $mine->completed_at ?? now(),
                    'updated_at' => now(),
                ]);
            });
        } catch (\Throwable $e) {
            Storage::disk('local')->delete($stored);
            throw $e;
        }

        foreach ($replaced as $p) {
            if (str_starts_with($p, 'form-registrations/'.$event->id.'/')) {
                Storage::disk('local')->delete($p);
            }
        }

        $this->alert(
            \App\Models\Institution::find($event->institution_id)?->user,
            'Data lanjutan masuk',
            ($mine->contact_name ?: $user->name).' melengkapi data untuk "'.$event->title.'".',
            '/panel/event/'.$event->id.'/registrasi'
        );

        return redirect('/events/'.$event->id.'/daftar?terkirim=1');
    }

    /** Notifikasi lonceng. Gagal kirim tidak boleh mengganggu pendaftaran. */
    private function alert(?object $to, string $title, string $message, string $url): void
    {
        try {
            $to?->notify(new AdminAlert($title, $message, $url));
        } catch (\Throwable $e) {
            report($e);
        }
    }

    private function guard(Event $event): void
    {
        abort_unless($event->status === Event::PUBLISHED && $event->registration_method !== 'ticket', 404);
    }

    private function categories(Event $event)
    {
        return DB::table('event_participant_categories')->where('event_id', $event->id)->orderBy('id')
            ->get(['id', 'name', 'competition_name', 'quota']);
    }

    private function active(Event $event): array
    {
        return FormFields::active($event, 'register');
    }

    private function used(Event $event, bool $isTeam, ?int $catId = null): int
    {
        if ($isTeam) {
            $q = DB::table('registration_teams as t')
                ->join('event_registrations as r', 'r.id', '=', 't.event_registration_id')
                ->where('r.event_id', $event->id)->whereNotIn('r.status', self::CLOSED);
            if ($catId) {
                $q->where('t.participant_category_id', $catId);
            }

            return $q->count();
        }
        $q = DB::table('event_registrations')->where('event_id', $event->id)->whereNotIn('status', self::CLOSED);
        if ($catId) {
            $q->where('participant_category_id', $catId);
        }

        return $q->count();
    }

    private function assertCapacity(Event $event, bool $isTeam, array $teams, ?int $catId, $cats): void
    {
        $want = $isTeam ? count($teams) : 1;
        $quota = (int) $event->participant_quota;
        if ($quota > 0 && $this->used($event, $isTeam) + $want > $quota) {
            throw ValidationException::withMessages(['form' => 'Kuota pendaftaran tidak mencukupi.']);
        }

        $perCat = $isTeam ? array_count_values(array_filter(array_column($teams, 'cat'))) : ($catId ? [$catId => 1] : []);
        foreach ($perCat as $id => $n) {
            $c = $cats->firstWhere('id', $id);
            if ($c && (int) $c->quota > 0 && $this->used($event, $isTeam, (int) $id) + $n > (int) $c->quota) {
                throw ValidationException::withMessages(['form' => 'Kuota kategori '.$c->name.' tidak mencukupi.']);
            }
        }
    }

    private function closedReason(Event $event, bool $isTeam): ?string
    {
        if ($event->registration_start && now()->lt(Carbon::parse($event->registration_start))) {
            return 'Pendaftaran belum dibuka.';
        }
        if ($event->registration_end) {
            $end = Carbon::parse($event->registration_end);
            if ($end->format('H:i:s') === '00:00:00') {
                $end = $end->endOfDay();
            }
            if (now()->gt($end)) {
                return 'Pendaftaran sudah ditutup.';
            }
        }
        $quota = (int) $event->participant_quota;
        if ($quota > 0 && $this->used($event, $isTeam) >= $quota) {
            return 'Kuota pendaftaran sudah penuh.';
        }

        return null;
    }

    /** Simpan berkas ke penyimpanan privat dengan nama acak. */
    private function keep(UploadedFile $file, Event $event, array &$stored): array
    {
        $ext = strtolower((string) ($file->guessExtension() ?: $file->getClientOriginalExtension()));
        $ext = $ext === 'jpeg' ? 'jpg' : $ext;
        abort_unless(in_array($ext, ['jpg', 'png', 'webp', 'pdf'], true), 422);
        $size = $file->getSize();
        $orig = Str::limit((string) preg_replace('/[^\w.\- ]+/u', '_', $file->getClientOriginalName()), 100, '');
        $path = $file->storeAs('form-registrations/'.$event->id, Str::random(40).'.'.$ext, 'local');
        $stored[] = $path;

        return ['path' => $path, 'name' => $orig, 'size' => $size];
    }
