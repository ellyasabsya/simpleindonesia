import { Toast } from '@/lib/toast';
import { Head, Link, router, usePage } from '@inertiajs/react';
import { ArrowLeft, Layers, LogOut, Menu, Moon, Sun, X } from 'lucide-react';
import { useState, type ReactNode } from 'react';
import PageBackground from '@/components/page-background';
import { eventMenuIcons } from '@/data/event-menu-icons';
import { panelMenu, type PanelMenuGroup } from '@/data/panel-menu';
import { useTheme } from '@/hooks/use-theme';

type PageProps = {
    auth?: { user?: { name?: string } };
    institution?: { name: string };
};

function Sidebar({ onNavigate }: { onNavigate?: () => void }) {
    const { url } = usePage();
    const path = url.split('?')[0];
    const { eventNav } = usePage<{ eventNav?: { title: string; items: { key: string; label: string; href: string }[]; parent?: { title: string; href: string }; subs?: { label: string; href: string }[] } }>().props;
    const shared = usePage().props as any;
    const menuPerms: string[] | null = shared.panelPerms ?? null;
    const menuOwner: boolean = !!shared.panelOwner;
    const MENU_NEED: Record<string, string[]> = {
        '/panel/event': ['events', 'checkin'],
        '/panel/ajukan': ['events'],
        '/panel/peserta': ['events', 'checkin'],
        '/panel/tiket': ['events', 'checkin', 'finance'],
        '/panel/voting': ['events'],
        '/panel/penilaian': ['events'],
        '/panel/pendapatan': ['finance', 'withdraw'],
    };
    const menuVisible = (href: string) => {
        if (menuOwner || menuPerms === null) return true;
        if (href === '/panel/profil') return false;
        const need = MENU_NEED[href];
        return !need || need.some((p) => menuPerms.includes(p));
    };
    const groups: PanelMenuGroup[] = eventNav
        ? [
              {
                  title: eventNav.title,
                  items: eventNav.items.filter((i) => !(eventNav.parent && i.href.endsWith('/kalender'))).map((i) => ({ label: i.label, href: i.href, icon: eventMenuIcons[i.key] ?? eventMenuIcons.dashboard })),
              },
              ...(eventNav.subs?.length
                  ? [{ title: 'Sub-event', items: eventNav.subs.map((s) => ({ label: s.label, href: s.href, icon: Layers })) }]
                  : []),
          ]
        : panelMenu;


    return (
        <div className="flex h-full flex-col">
            <Link href="/" onClick={onNavigate} className="flex items-center gap-3 px-6 pb-5 pt-6">
                <span className="flex h-10 w-10 items-center justify-center rounded-xl bg-gradient-to-br from-blue-600 to-indigo-900 text-xl font-black text-white">
                    S
                </span>
                <span>
                    <span className="block text-lg font-extrabold leading-none tracking-[0.2em] text-indigo-900 dark:text-white">SIMPLE</span>
                    <span className="mt-1 block text-[10px] font-bold uppercase tracking-[0.25em] text-indigo-500 dark:text-indigo-300">Panel Lembaga</span>
                </span>
            </Link>

            <nav className="flex-1 space-y-5 overflow-y-auto px-4 pb-4 [scrollbar-width:none]">
                {eventNav?.parent && (
                    <Link href={eventNav.parent.href} onClick={onNavigate} className="flex items-center gap-2 rounded-xl bg-indigo-50 px-3 py-2.5 text-sm font-bold text-indigo-800 transition hover:bg-indigo-100 dark:bg-indigo-500/15 dark:text-indigo-100">
                        <ArrowLeft className="h-4 w-4" /> {eventNav.parent.title}
                    </Link>
                )}
                {groups.map((g0) => ({ ...g0, items: g0.items.filter((it) => menuVisible(it.href)) })).filter((g0) => g0.items.length > 0).map((group) => (
                    <div key={group.title}>
                        <p className="mb-2 px-3 text-[11px] font-bold uppercase tracking-[0.2em] text-slate-400 dark:text-slate-500">{group.title}</p>
                        <div className="space-y-1">
                            {group.items.map(({ label, href, icon: Icon }) => {
                                const active = href === '/panel' || eventNav ? path === href : path.startsWith(href);
                                return (
                                    <Link
                                        key={href}
                                        href={href}
                                        onClick={onNavigate}
                                        className={`flex items-center gap-3 rounded-xl px-3 py-2.5 text-sm font-semibold transition ${
                                            active
                                                ? 'bg-gradient-to-r from-indigo-700 via-blue-600 to-violet-600 text-white shadow-md shadow-indigo-600/25'
                                                : 'text-slate-600 hover:bg-white/80 hover:text-indigo-800 dark:text-slate-300 dark:hover:bg-white/10 dark:hover:text-white'
                                        }`}
                                    >
                                        <Icon className="h-[18px] w-[18px]" />
                                        {label}
                                    </Link>
                                );
                            })}
                        </div>
                    </div>
                ))}
            </nav>

            <div className="border-t border-slate-200/70 p-4 dark:border-white/10">
                <Link
                    href="/panel"
                    onClick={onNavigate}
                    className="flex items-center justify-center gap-2 rounded-xl bg-white/80 py-2.5 text-sm font-semibold text-slate-700 shadow-sm transition hover:bg-white hover:text-indigo-700 dark:bg-white/10 dark:text-slate-200 dark:hover:bg-white/20"
                >
                    <ArrowLeft className="h-4 w-4" />
                    Panel Lembaga
                </Link>
            </div>
        </div>
    );
}

<?php

namespace App\Http\Controllers\Panel;

use App\Http\Controllers\Controller;
use App\Models\Event;
use App\Support\RegistrationForm;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Carbon;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Str;
use Illuminate\Validation\Rule;
use Illuminate\Validation\ValidationException;
use Symfony\Component\HttpFoundation\StreamedResponse;

class RegistrationFormController extends Controller
{
    private function own(Request $request, Event $event): void
    {
        $inst = $request->attributes->get('institution');
        abort_unless($inst && $event->institution_id === $inst->id, 404);
        abort_if($event->registration_method === 'ticket', 404);
    }

    public function update(Request $request, Event $event): RedirectResponse
    {
        $this->own($request, $event);

        $d = $request->validate([
            'fields' => ['present', 'array', 'max:200'],
            'fields.*.key' => ['nullable', 'string', 'max:20'],
            'fields.*.label' => ['required', 'string', 'max:120'],
            'fields.*.type' => ['required', Rule::in(RegistrationForm::TYPES)],
            'fields.*.required' => ['boolean'],
            'fields.*.stage' => ['nullable', Rule::in(['register', 'after'])],
            'fields.*.options' => ['nullable', 'array', 'max:50'],
            'fields.*.options.*' => ['string', 'max:80'],
            'message' => ['nullable', 'string', 'max:2000'],
        ], [], ['fields.*.label' => 'Label kolom']);

        $existing = collect(\App\Support\FormFields::all($event))->keyBy('key');
        $taken = $existing->keys()->all();
        $out = [];
        $seen = [];

        foreach ($d['fields'] as $i => $f) {
            $type = $f['type'];
            $options = [];
            if (RegistrationForm::needsOptions($type)) {
                $options = array_values(array_unique(array_filter(
                    array_map('trim', $f['options'] ?? []),
                    fn ($o) => $o !== ''
                )));
                if (! $options) {
                    throw ValidationException::withMessages([
                        "fields.$i.options" => 'Kolom "'.trim($f['label']).'" butuh minimal satu pilihan.',
                    ]);
                }
            }

            $key = $f['key'] ?? null;
            if (! $key || ! $existing->has($key) || isset($seen[$key])) {
                do {
                    $key = 'f_'.Str::lower(Str::random(8));
                } while (in_array($key, $taken, true) || isset($seen[$key]));
            }
            $seen[$key] = true;

            $out[] = [
                'key' => $key,
                'label' => trim($f['label']),
                'type' => $type,
                'required' => (bool) ($f['required'] ?? false),
                'stage' => (($f['stage'] ?? 'register') === 'after') ? 'after' : 'register',
                'options' => $options,
                'archived' => false,
            ];
        }

        // Kolom yang dihapus tapi sudah punya jawaban diarsipkan, supaya jawaban lama tidak hilang.
        if (DB::table('event_registrations')->where('event_id', $event->id)->where(fn ($q) => $q->whereNotNull('answers')->orWhereNotNull('followup_answers'))->exists()) {
            foreach ($existing as $key => $f) {
                if (! isset($seen[$key])) {
                    $out[] = ['archived' => true] + $f;
                }
            }
        }

        DB::table('events')->where('id', $event->id)->update([
            'registration_fields' => json_encode($out, JSON_UNESCAPED_UNICODE),
            'registration_message' => $d['message'] ?? null,
        ]);

        return back();
    }

    public function review(Request $request, Event $event, int $reg): RedirectResponse
    {
        $this->own($request, $event);

        $d = $request->validate([
            'status' => ['required', Rule::in(['menunggu', 'terdaftar', 'ditolak'])],
            'note' => ['nullable', 'string', 'max:500'],
        ]);
        if ($d['status'] === 'ditolak' && blank($d['note'] ?? null)) {
            throw ValidationException::withMessages(['note' => 'Alasan penolakan wajib diisi.']);
        }

        $n = DB::table('event_registrations')->where('event_id', $event->id)->where('id', $reg)->update([
            'status' => $d['status'],
            'review_note' => $d['status'] === 'ditolak' ? trim($d['note']) : null,
            'reviewed_at' => now(),
            'updated_at' => now(),
        ]);
        abort_unless($n, 404);

        if ($d['status'] !== 'menunggu') {
            try {
                $row = DB::table('event_registrations')->where('id', $reg)->first();
                $to = $row ? \App\Models\User::find($row->user_id) : null;
                if ($to) {
                    $ok = $d['status'] === 'terdaftar';
                    $needMore = $ok && $event->competition_type !== 'team' && count(\App\Support\FormFields::active($event, 'after')) > 0;
                    $to->notify(new \App\Notifications\AdminAlert(
                        $ok ? 'Pendaftaran disetujui' : 'Pendaftaran ditolak',
                        $ok
                            ? 'Pendaftaran Anda di "'.$event->title.'" disetujui.'.($needMore ? ' Panitia meminta data tambahan, silakan dilengkapi.' : '')
                            : 'Pendaftaran Anda di "'.$event->title.'" ditolak. Alasan: '.trim($d['note']),
                        '/events/'.$event->id.'/daftar'
                    ));
                }
            } catch (\Throwable $e) {
                report($e);
            }
        }

        return back();
    }

    public function destroy(Request $request, Event $event, int $reg): RedirectResponse
    {
        $this->own($request, $event);

        $row = DB::table('event_registrations')->where('event_id', $event->id)->where('id', $reg)->first();
        if ($row) {
            $paths = [];
            foreach (array_merge(json_decode((string) $row->answers, true) ?: [], json_decode((string) $row->followup_answers, true) ?: []) as $v) {
                if (is_array($v) && isset($v['path'])) {
                    $paths[] = $v['path'];
                }
            }
            if ($row->payment_proof) {
                $paths[] = $row->payment_proof;
            }
            foreach ($paths as $p) {
                if (str_starts_with($p, 'form-registrations/'.$event->id.'/')) {
                    Storage::disk('local')->delete($p);
                }
            }
            DB::table('registration_teams')->where('event_registration_id', $row->id)->delete();
            DB::table('event_registrations')->where('id', $row->id)->delete();
        }

        return back();
    }

    public function export(Request $request, Event $event): StreamedResponse
    {
        $this->own($request, $event);
        $isTeam = $event->competition_type === 'team';
        $fields = $isTeam ? [] : \App\Support\FormFields::all($event);
        $tz = config('app.agenda_timezone', 'Asia/Jakarta');
        $cats = DB::table('event_participant_categories')->where('event_id', $event->id)->pluck('name', 'id');
        // Cegah rumus spreadsheet dari isian pendaftar.
        $safe = fn ($v) => is_string($v) && $v !== '' && str_contains("=+-@\t\r", $v[0]) ? "'".$v : $v;
        $name = 'pendaftar-'.Str::slug($event->title ?: 'event').'.csv';

        return response()->streamDownload(function () use ($event, $fields, $tz, $cats, $safe) {
            $out = fopen('php://output', 'w');
            fwrite($out, "\xEF\xBB\xBF");
            fputcsv($out, array_merge(
                ['Nama', 'Email', 'WhatsApp', 'Sekolah / Instansi', 'Kategori', 'Tim', 'Status', 'Total tagihan', 'Waktu daftar'],
                array_map(fn ($f) => $safe($f['label']), $fields)
            ), ',', '"', '\\');

            DB::table('event_registrations')->where('event_id', $event->id)->orderBy('id')->chunk(500, function ($chunk) use ($out, $fields, $tz, $cats, $safe) {
                $users = DB::table('users')->whereIn('id', $chunk->pluck('user_id'))->get(['id', 'name', 'email'])->keyBy('id');
                $teams = DB::table('registration_teams')->whereIn('event_registration_id', $chunk->pluck('id'))->orderBy('id')->get()->groupBy('event_registration_id');
                foreach ($chunk as $r) {
                    $u = $users->get($r->user_id);
                    $a = array_merge(json_decode((string) $r->answers, true) ?: [], json_decode((string) $r->followup_answers, true) ?: []);
                    $row = [
                        $r->contact_name ?: ($u->name ?? ''),
                        $r->contact_email ?: ($u->email ?? ''),
                        $r->contact_phone,
                        $r->school,
                        $cats->get($r->participant_category_id),
                        $teams->get($r->id, collect())->map(fn ($t) => $t->name.($cats->get($t->participant_category_id) ? ' ('.$cats->get($t->participant_category_id).')' : ''))->implode('; '),
                        $r->status,
                        (int) $r->total_fee,
                        Carbon::parse($r->created_at)->setTimezone($tz)->format('Y-m-d H:i'),
                    ];
                    foreach ($fields as $f) {
                        $v = $a[$f['key']] ?? '';
                        if (is_array($v) && isset($v['path'])) {
                            $v = $v['name'] ?? 'berkas';
                        }
                        $row[] = is_array($v) ? implode(', ', $v) : $v;
                    }
                    fputcsv($out, array_map($safe, $row), ',', '"', '\\');
                }
            });
            fclose($out);
        }, $name, ['Content-Type' => 'text/csv; charset=UTF-8']);
    }

    public function berkas(Request $request, Event $event, int $reg, string $key)
    {
        $this->own($request, $event);
        $r = DB::table('event_registrations')->where('event_id', $event->id)->where('id', $reg)->first();
        abort_unless($r, 404);

        $v = array_merge(json_decode((string) $r->answers, true) ?: [], json_decode((string) $r->followup_answers, true) ?: [])[$key] ?? null;
        abort_unless(is_array($v) && isset($v['path']) && str_starts_with($v['path'], 'form-registrations/'.$event->id.'/'), 404);

        $disk = Storage::disk('local');
        abort_unless($disk->exists($v['path']), 404);

        return $disk->response($v['path'], $v['name'] ?? basename($v['path']), ['X-Content-Type-Options' => 'nosniff']);
    }

    public function bukti(Request $request, Event $event, int $reg)
    {
        $this->own($request, $event);
        $r = DB::table('event_registrations')->where('event_id', $event->id)->where('id', $reg)->first();
        abort_unless($r && $r->payment_proof && str_starts_with($r->payment_proof, 'form-registrations/'.$event->id.'/'), 404);

        $disk = Storage::disk('local');
        abort_unless($disk->exists($r->payment_proof), 404);

        return $disk->response($r->payment_proof, 'bukti-transfer-'.$r->id.'.'.pathinfo($r->payment_proof, PATHINFO_EXTENSION), ['X-Content-Type-Options' => 'nosniff']);
    }
}
